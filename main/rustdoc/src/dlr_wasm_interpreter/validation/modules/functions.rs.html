30/// Decodes a code section[^binary-format] and validates it[^validation].
31///
32/// [^binary-format]: [WebAssembly Specification 2.0 - 5.5.13. Code Section](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#binary-codesec).
33/// [^validation]: [WebAssembly Specification 2.0 - 3.4.1. Functions](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#functions%E2%91%A4).
34///
35/// # Safety
36///
37/// The caller must ensure that all index values passed into this function are
38/// valid in the relevant `IdxVec`. The following table lists all index types
39/// and their respective `IdxVec` types:
40///
41/// | Index | `IdxVec` |
42/// | ----- | -------- |
43/// | [`TypeIdx`] | [`IdxVec<TypeIdx, FuncType>`] |
44/// | [`FuncIdx`] | [`IdxVec<FuncIdx, TypeIdx>`] contained in [`ExtendedIdxVec<FuncIdx, TypeIdx>`] |
45/// | [`TableIdx`] | [`IdxVec<TableIdx, TableType>`] |
46#[allow(clippy::too_many_arguments)]
47pub unsafe fn decode_and_validate_code_section<T2: ValidationConfig>(
48    wasm: &mut WasmDecoder,
49    fn_types: &IdxVec<TypeIdx, FuncType>,
50    c_funcs: &ExtendedIdxVec<FuncIdx, TypeIdx>,
51    c_globals: &IdxVec<GlobalIdx, Global>,
52    c_mems: &IdxVec<MemIdx, MemType>,
53    data_count: Option<u32>,
54    c_tables: &IdxVec<TableIdx, TableType>,
55    c_elems: &IdxVec<ElemIdx, ElemType>,
56    validation_context_refs: &BTreeSet<FuncIdx>,
57    sidetable: &mut Sidetable,
58    user_data: &mut T2,
59) -> Result<Vec<(Span, usize)>, ValidationError> {
60    let code_block_spans_stps = wasm.decode_vec_enumerated(|wasm, idx| {
61        // We need to offset the index by the number of functions that were
62        // imported. Imported functions always live at the start of the index
63        // space.
64        let ty_idx = c_funcs
65            .iter_local_definitions()
66            .nth(idx.into_usize())
67            .ok_or(ValidationError::FunctionAndCodeSectionsHaveDifferentLengths)?;
68
69        // SAFETY: The caller ensures that all passed `TypeIdx` values,
70        // including this one, are valid in this `IdxVec<TypeIdx, FuncType>`.
71        let func_ty: FuncType = unsafe { fn_types.get(*ty_idx).clone() };
72
73        let func_size = wasm.decode_var_u32()?;
74        let func_block = wasm.make_span(func_size.into_usize())?;
75        let previous_pc = wasm.pc;
76
77        let locals = {
78            let params = func_ty.params.valtypes.iter().cloned();
79            let declared_locals = decode_locals(wasm)?;
80            params.chain(declared_locals).collect::<Vec<ValType>>()
81        };
82
83        let mut stack = ValidationStack::new_for_func(func_ty);
84        let stp = sidetable.len();
85
86        // SAFETY: The caller ensures the same safety requirements for the same
87        // unmodified index values.
88        unsafe {
89            decode_and_validate_expr(
90                wasm,
91                &mut stack,
92                sidetable,
93                &locals,
94                c_globals,
95                fn_types,
96                c_funcs.inner(),
97                c_mems,
98                data_count,
99                c_tables,
100                c_elems,
101                validation_context_refs,
102                user_data,
103            )
104        }?;
105
106        // Check if there were unread trailing instructions after the last END
107        if previous_pc + func_size.into_usize() != wasm.pc {
108            return Err(ValidationError::CodeExprHasTrailingInstructions);
109        }
110
111        Ok((func_block, stp))
112    })?;
113
114    trace!(
115        "Read code section. Found {} code blocks",
116        code_block_spans_stps.len()
117    );
118
119    Ok(code_block_spans_stps)
120}
