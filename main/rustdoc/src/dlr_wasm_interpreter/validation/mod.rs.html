42/// Information collected from validating a module.
43///
44/// This can be used to instantiate a new module instance in some
45/// [`Store`](crate::Store) thorugh
46/// [`Store::module_instantiate`](crate::Store::module_instantiate)
47#[derive(Clone, Debug)]
48pub struct Module<'bytecode> {
49    pub(crate) wasm: &'bytecode [u8],
50    pub(crate) types: IdxVec<TypeIdx, FuncType>,
51    pub(crate) imports: Vec<Import<'bytecode>>,
52    pub(crate) functions: ExtendedIdxVec<FuncIdx, TypeIdx>,
53    pub(crate) tables: ExtendedIdxVec<TableIdx, TableType>,
54    pub(crate) memories: ExtendedIdxVec<MemIdx, MemType>,
55    pub(crate) globals: ExtendedIdxVec<GlobalIdx, Global>,
56    pub(crate) exports: Vec<Export<'bytecode>>,
57    pub(crate) elements: IdxVec<ElemIdx, ElemType>,
58    pub(crate) data: IdxVec<DataIdx, DataSegment>,
59    /// Each block contains the validated code section and the stp corresponding to
60    /// the beginning of that code section
61    pub(crate) func_blocks_stps: Vec<(Span, usize)>,
62    pub(crate) sidetable: Sidetable,
63    /// The start function which is automatically executed during instantiation
64    pub(crate) start: Option<FuncIdx>,
65    pub(crate) custom_sections: Vec<CustomSection<'bytecode>>,
66    // pub(crate) exports_length: Exported,
67}

69fn validate_no_duplicate_exports(module: &Module) -> Result<(), ValidationError> {
70    let mut found_export_names: btree_set::BTreeSet<&str> = btree_set::BTreeSet::new();
71    for export in &module.exports {
72        if found_export_names.contains(export.name) {
73            return Err(ValidationError::DuplicateExportName);
74        }
75        found_export_names.insert(export.name);
76    }
77    Ok(())
78}

80pub fn decode_and_validate<'wasm, T: ValidationConfig>(
81    wasm: &'wasm [u8],
82    user_data: &mut T,
83) -> Result<Module<'wasm>, ValidationError> {
84    let mut wasm = WasmDecoder::new(wasm);
85
86    // represents C.refs in https://webassembly.github.io/spec/core/valid/conventions.html#context
87    // A func.ref instruction is onlv valid if it has an immediate that is a member of C.refs.
88    // this list holds all the func_idx's occurring in the module, except in its functions or start function.
89    // I make an exception here by not including func_idx's occuring within data segments in C.refs as well, so that single pass validation is possible.
90    // If there is a func_idx within the data segment, this would ultimately mean that data segment cannot be validated,
91    // therefore this hack is acceptable.
92    // https://webassembly.github.io/spec/core/valid/modules.html#data-segments
93    // https://webassembly.github.io/spec/core/valid/modules.html#valid-module
94
95    let mut validation_context_refs: BTreeSet<FuncIdx> = BTreeSet::new();
96
97    let [0x00, 0x61, 0x73, 0x6d] = wasm.strip_bytes::<4>()? else {
98        return Err(DecodingError::InvalidMagic.into());
99    };
100
101    let [0x01, 0x00, 0x00, 0x00] = wasm.strip_bytes::<4>()? else {
102        return Err(DecodingError::InvalidBinaryFormatVersion.into());
103    };
104
105    let mut custom_sections = Vec::new();
106    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
107
108    let types = decode_section_if_ty_matches::<_, DecodingError>(&mut wasm, SectionTy::Type, |wasm, _| {
109        let collected= wasm.decode_vec_map(FuncType::decode).and_then(Iterator::collect)?;
110        let idx_vec = IdxVec::new(collected).expect("that index space creation never fails because the length of the types vector is encoded as a 32-bit integer in the bytecode");
111        Ok(idx_vec)
112    })?.unwrap_or_default();
113
114    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
115
116    let imports = decode_section_if_ty_matches(&mut wasm, SectionTy::Import, |wasm, _| {
117        let elements = wasm.decode_vec_map(|wasm| Import::decode_and_validate(wasm, &types))?;
118        elements.collect::<Result<Vec<Import>, ValidationError>>()
119    })?
120    .unwrap_or_default();
121
122    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
123
124    // The `Function` section only covers module-level (or "local") functions.
125    // Imported functions have their types known in the `import` section. Both
126    // local and imported functions share the same index space.
127    //
128    // Imported functions are given priority and have the first indicies, and
129    // only after that do the local functions get assigned their indices.
130    let local_functions =
131        decode_section_if_ty_matches(&mut wasm, SectionTy::Function, |wasm, _| {
132            wasm.decode_vec_map(|wasm| TypeIdx::decode_and_validate(wasm, &types))
133                .map_err(ValidationError::from)
134                .and_then(Iterator::collect)
135        })?
136        .unwrap_or_default();
137
138    let imported_functions = imports.iter().filter_map(|import| match &import.desc {
139        ImportDesc::Func(type_idx) => Some(*type_idx),
140        _ => None,
141    });
142
143    let functions = ExtendedIdxVec::new(imported_functions.collect(), local_functions)
144        .map_err(|IdxVecOverflowError| ValidationError::TooManyFunctions)?;
145
146    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
147
148    let imported_tables = imports.iter().filter_map(|m| match m.desc {
149        ImportDesc::Table(table) => Some(table),
150        _ => None,
151    });
152    let local_tables = decode_section_if_ty_matches(&mut wasm, SectionTy::Table, |wasm, _| {
153        wasm.decode_vec_map(TableType::decode_and_validate)
154            .map_err(ValidationError::from)
155            .and_then(Iterator::collect)
156    })?
157    .unwrap_or_default();
158
159    let tables = ExtendedIdxVec::new(imported_tables.collect(), local_tables)
160        .map_err(|IdxVecOverflowError| ValidationError::TooManyTables)?;
161
162    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
163
164    let imported_memories = imports.iter().filter_map(|m| match m.desc {
165        ImportDesc::Mem(mem) => Some(mem),
166        _ => None,
167    });
168    // let imported_memories_length = imported_memories.len();
169    let local_memories = decode_section_if_ty_matches(&mut wasm, SectionTy::Memory, |wasm, _| {
170        wasm.decode_vec_map(MemType::decode_and_validate)
171            .map_err(ValidationError::from)
172            .and_then(Iterator::collect)
173    })?
174    .unwrap_or_default();
175
176    let memories = ExtendedIdxVec::new(imported_memories.collect(), local_memories)
177        .map_err(|IdxVecOverflowError| ValidationError::TooManyMemories)?;
178
179    if memories.inner().len() > 1 {
180        return Err(ValidationError::UnsupportedMultipleMemoriesProposal);
181    }
182
183    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
184
185    let imported_global_types: Vec<GlobalType> = imports
186        .iter()
187        .filter_map(|m| match m.desc {
188            ImportDesc::Global(global) => Some(global),
189            _ => None,
190        })
191        .collect();
192    let local_globals = decode_section_if_ty_matches(&mut wasm, SectionTy::Global, |wasm, _| {
193        wasm.decode_vec_map(|wasm| {
194            Global::decode_and_validate(
195                wasm,
196                &imported_global_types,
197                &mut validation_context_refs,
198                functions.inner(),
199            )
200        })
201        .map_err(ValidationError::from)
202        .and_then(Iterator::collect)
203    })?
204    .unwrap_or_default();
205
206    let imported_globals = imported_global_types.iter().map(|ty| Global {
207        // TODO using a default MAX value for spans that are never executed is
208        // not really safe. Maybe opt for an Option instead.
209        init_expr: Span::new(usize::MAX, 0),
210        ty: *ty,
211    });
212    let globals = ExtendedIdxVec::new(imported_globals.collect(), local_globals)
213        .map_err(|IdxVecOverflowError| ValidationError::TooManyGlobals)?;
214
215    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
216
217    let exports: Vec<Export> =
218        decode_section_if_ty_matches(&mut wasm, SectionTy::Export, |wasm, _| {
219            wasm.decode_vec_map(|wasm| {
220                Export::decode_and_validate(
221                    wasm,
222                    functions.inner(),
223                    tables.inner(),
224                    memories.inner(),
225                    globals.inner(),
226                )
227            })
228            .map_err(ValidationError::from)
229            .and_then(Iterator::collect)
230        })?
231        .unwrap_or_default();
232    validation_context_refs.extend(exports.iter().filter_map(
233        |Export { name: _, desc }| match desc {
234            ExportDesc::Func(func_idx) => Some(func_idx),
235            _ => None,
236        },
237    ));
238
239    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
240
241    let start = decode_section_if_ty_matches(&mut wasm, SectionTy::Start, |wasm, _| {
242        let func_idx = FuncIdx::decode_and_validate(wasm, functions.inner())?;
243
244        // start function signature must be [] -> []
245        // https://webassembly.github.io/spec/core/valid/modules.html#start-function
246        // SAFETY: We just validated this function index using the same
247        // `IdxVec`.
248        let type_idx = unsafe { functions.inner().get(func_idx) };
249
250        // SAFETY: There exists only one `IdxVec<TypeIdx, FuncType>` in the
251        // current function. Therefore, this has to be the same one used to
252        // create and validate this `TypeIdx`.
253        let func_type = unsafe { types.get(*type_idx) };
254        if !func_type.is_empty() {
255            Err(ValidationError::InvalidStartFunctionSignature)
256        } else {
257            Ok(func_idx)
258        }
259    })?;
260
261    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
262
263    let elements = decode_section_if_ty_matches::<_, ValidationError>(&mut wasm, SectionTy::Element, |wasm, _| {
264        let collected = wasm.decode_vec_map(|wasm| {
265            ElemType::decode_and_validate(
266            wasm,
267            functions.inner(),
268            &mut validation_context_refs,
269            tables.inner(),
270            &imported_global_types,
271        )}).map_err(ValidationError::from).and_then(Iterator::collect)?;
272        let idx_vec = IdxVec::new(collected).expect("that index space creation never fails because the length of the elements vector is encoded as a 32-bit integer in the bytecode");
273        Ok(idx_vec)
274    })?
275    .unwrap_or_default();
276
277    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
278
279    // https://webassembly.github.io/spec/core/binary/modules.html#data-count-section
280    // As per the official documentation:
281    //
282    // The data count section is used to simplify single-pass validation. Since the data section occurs after the code section, the `memory.init` and `data.drop` and instructions would not be able to check whether the data segment index is valid until the data section is read. The data count section occurs before the code section, so a single-pass validator can use this count instead of deferring validation.
283    let data_count: Option<u32> =
284        decode_section_if_ty_matches(&mut wasm, SectionTy::DataCount, |wasm, _| {
285            wasm.decode_var_u32()
286        })?;
287
288    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
289
290    let mut sidetable = Sidetable::new();
291    let func_blocks_stps = decode_section_if_ty_matches(&mut wasm, SectionTy::Code, |wasm, _| {
292        // SAFETY: It is required that all passed index values are valid in all
293        // passed `IdxVec`s. The current function does not take any index types
294        // as arguments and every `IdxVec<..., ...>` is unique because they use
295        // different generics. Therefore, all index types must be valid in their
296        // relevant `IdxVec`s.
297        unsafe {
298            decode_and_validate_code_section(
299                wasm,
300                &types,
301                &functions,
302                globals.inner(),
303                memories.inner(),
304                data_count,
305                tables.inner(),
306                &elements,
307                &validation_context_refs,
308                &mut sidetable,
309                user_data,
310            )
311        }
312    })?
313    .unwrap_or_default();
314
315    if func_blocks_stps.len() != functions.len_local_definitions().into_usize() {
316        return Err(ValidationError::FunctionAndCodeSectionsHaveDifferentLengths);
317    }
318
319    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
320
321    let data_section = decode_section_if_ty_matches::<_, ValidationError>(&mut wasm, SectionTy::Data, |wasm, _| {
322        let collected = wasm.decode_vec_map(|wasm| {
323            DataSegment::decode_and_validate(wasm, &imported_global_types, functions.inner(), memories.inner())
324        }).map_err(ValidationError::from).and_then(Iterator::collect)?;
325        let idx_vec = IdxVec::new(collected).expect("that index space creation never fails because the length of the data segments vector is encoded as a 32-bit integer in the bytecode");
326        Ok(idx_vec)
327    })?
328    .unwrap_or_default();
329
330    // https://webassembly.github.io/spec/core/binary/modules.html#data-count-section
331    if let Some(data_count) = data_count {
332        if data_count != data_section.len() {
333            return Err(ValidationError::DataCountAndDataSectionsLengthAreDifferent);
334        }
335    }
336
337    read_all_custom_sections(&mut wasm, &mut custom_sections)?;
338
339    // All sections should have been handled
340    if !wasm.remaining_bytes().is_empty() {
341        let remaining_section_ty = SectionTy::decode(&mut wasm).expect(
342            "that the section type is not malformed, because it must have been peeked before",
343        );
344        return Err(DecodingError::SectionOutOfOrder(remaining_section_ty).into());
345    }
346
347    let module = Module {
348        wasm: wasm.into_inner(),
349        types,
350        imports,
351        functions,
352        tables,
353        memories,
354        globals,
355        exports,
356        func_blocks_stps,
357        sidetable,
358        data: data_section,
359        start,
360        elements,
361        custom_sections,
362    };
363    validate_no_duplicate_exports(&module)?;
364
365    Ok(module)
366}

368/// Reads the next sections as long as they are custom sections and pushes them
369/// into the `custom_sections` vector.
370fn read_all_custom_sections<'wasm>(
371    wasm: &mut WasmDecoder<'wasm>,
372    custom_sections: &mut Vec<CustomSection<'wasm>>,
373) -> Result<(), ValidationError> {
374    while let Some(custom_section) =
375        decode_section_if_ty_matches(wasm, SectionTy::Custom, CustomSection::decode)?
376    {
377        custom_sections.push(custom_section);
378    }
379
380    Ok(())
381}

383impl<'wasm> Module<'wasm> {
384    /// Returns the imports of this module as an iterator. Each import consist
385    /// of a module name, a name and an extern type.
386    ///
387    /// See: WebAssembly Specification 2.0 - 7.1.5 - module_imports
388    pub fn imports<'a>(
389        &'a self,
390    ) -> Map<
391        core::slice::Iter<'a, Import<'wasm>>,
392        impl FnMut(&'a Import<'wasm>) -> (&'a str, &'a str, ExternType),
393    > {
394        self.imports.iter().map(|import| {
395            // SAFETY: This is sound because the argument is `self` and the
396            // import desc also comes from `self`.
397            let extern_type = unsafe { import.desc.extern_type_owned(self) };
398            (import.module_name, import.name, extern_type)
399        })
400    }
401
402    /// Returns the exports of this module as an iterator. Each export consist
403    /// of a name, and an extern type.
404    ///
405    /// See: WebAssembly Specification 2.0 - 7.1.5 - module_exports
406    pub fn exports<'a>(
407        &'a self,
408    ) -> Map<
409        core::slice::Iter<'a, Export<'wasm>>,
410        impl FnMut(&'a Export<'wasm>) -> (&'a str, ExternType),
411    > {
412        self.exports.iter().map(|export| {
413            // SAFETY: This is sound because the argument is `self` and the
414            // export desc also comes from `self`.
415            let extern_type = unsafe { export.desc.extern_type(self) };
416            (export.name, extern_type)
417        })
418    }
419
420    /// Returns a list of all custom sections in the bytecode. Every custom
421    /// section consists of its name and the custom section's bytecode
422    /// (excluding the name itself).
423    pub fn custom_sections(&self) -> &[CustomSection<'wasm>] {
424        &self.custom_sections
425    }
