17/// Decodes a constant expression[^binary-format] and validates it[^validation].
18///
19/// [^binary-format]: [WebAssembly Specification 2.0 - 5.4.9. Expressions](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#binary-expr).
20/// [^validation]: [WebAssembly Specification 2.0 - 3.3.10. Expressions](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#expressions%E2%91%A2).
21///
22/// This function is used to validate that a constant expression produces the expected result. The main use case for
23/// this is to validate that an initialization expression for a global returns the correct value.
24///
25/// Note: to be valid, constant expressions may not leave garbage data on the stack. It may leave only what is expected
26/// and nothing more.
27///
28/// Valid constant instructions are:
29/// - Core: <https://webassembly.github.io/spec/core/valid/instructions.html#valid-constant>
30/// - Extended Proposal: <https://webassembly.github.io/extended-const/core/valid/instructions.html#valid-constant>
31///
32/// # The Wonders of `global.get`
33/// The `global.get` instruction is quite picky by nature. To make a long story short, there are two rules to follow to
34/// be able to use this expression.
35///
36/// ## 1. The referenced global must be imported
37/// Take the example code:
38/// ```wat
39/// (module
40///     (global (export "g") (mut i32) (
41///         i32.add (i32.const 1) (i32.const 2)
42///     ))
43///
44///     (global (export "h1") i32 (
45///         i32.const 1
46///     ))
47///
48///     (global (export "h2") i32 (
49///         global.get 1
50///     ))
51///
52///     (func (export "f")
53///         i32.const 100
54///         global.set 0))
55/// ```
56///
57/// When compiling with wat2wasm, the following error is thrown:
58/// ```wat
59/// Error: validate failed:
60/// test.wast:11:24: error: initializer expression can only reference an imported global
61///             global.get 1
62///                        ^
63/// ```
64///
65/// When compiling the code with the latest dev build of wasmtime, the following error is thrown:
66/// ```wat
67/// failed to parse WebAssembly module
68///
69/// Caused by:
70///     constant expression required: global.get of locally defined global (at offset 0x24)
71/// ```
72///
73/// ## 2. The referenced global must be immutable
74///
75///```wat
76/// (module
77///     (import "env" "g" (global (mut i32)))
78///     (global (export "h") (mut i32) (
79///         i32.add (i32.const 1) (global.get 0)
80///     ))
81///   )
82/// ```
83///
84/// When compiling with wat2wasm, the following error is thrown:
85/// ```wat
86/// Error: validate failed:
87/// test.wast:4:27: error: initializer expression cannot reference a mutable global
88///     i32.add (i32.const 1) (global.get 0)
89/// ```
90///
91/// # Note
92/// The following instructions are not yet supported:
93/// - `ref.null`
94/// - `ref.func`
95/// - `global.get`
96pub fn decode_and_validate_constant_expression(
97    wasm: &mut WasmDecoder,
98    stack: &mut ValidationStack,
99    // The globals slice should contain ONLY imported globals IF AND ONLY IF we are calling `read_constant_expression` for local globals instantiation
100    // As per https://webassembly.github.io/spec/core/valid/modules.html (bottom of the page):
101    //
102    //  Globals, however, are not recursive and not accessible within constant expressions when they are defined locally. The effect of defining the limited context C'
103    //   for validating certain definitions is that they can only access functions and imported globals and nothing else.
104    imported_globals: &[GlobalType],
105    c_funcs: &IdxVec<FuncIdx, TypeIdx>,
106) -> Result<(Span, Vec<FuncIdx>), ValidationError> {
107    let start_pc = wasm.pc;
108    let mut seen_func_idxs: Vec<FuncIdx> = Vec::new();
109
110    loop {
111        let Ok(first_instr_byte) = wasm.decode_u8() else {
112            return Err(ValidationError::ExprMissingEnd);
113        };
114
115        trace!(
116            "Validating const instruction {} at pc={}",
117            instruction_byte_to_str(first_instr_byte),
118            wasm.pc
119        );
120
121        use crate::core::structure::instructions::*;
122        match first_instr_byte {
123            END => {
124                // The code here for checking the global type was moved to where the global is actually validated
125                return Ok((Span::new(start_pc, wasm.pc - start_pc), seen_func_idxs));
126            }
127            GLOBAL_GET => {
128                // Unfortunately, we cannot use the GlobalIdx type yet, because
129                // constant expressions may only access imported globals.  Wasm
130                // specifies that only imported globals may be accessed (see
131                // comment on `imported_globals` parameter).
132                let imported_global_idx = wasm.decode_var_u32()?;
133
134                let global = imported_globals
135                    .get(imported_global_idx.into_usize())
136                    .ok_or(ValidationError::InvalidGlobalIdx(imported_global_idx))?;
137
138                if global.is_mut {
139                    return Err(ValidationError::MutGlobalInConstGlobalGet);
140                }
141
142                stack.push_valtype(global.ty);
143            }
144            I32_CONST => {
145                let _num = wasm.decode_var_i32()?;
146                stack.push_valtype(ValType::NumType(NumType::I32));
147            }
148            F32_CONST => {
149                let _num = wasm.decode_f32();
150                stack.push_valtype(ValType::NumType(NumType::F32));
151            }
152            F64_CONST => {
153                let _num = wasm.decode_f64();
154                stack.push_valtype(ValType::NumType(NumType::F64));
155            }
156            I64_CONST => {
157                let _num = wasm.decode_var_i64()?;
158                stack.push_valtype(ValType::NumType(NumType::I64));
159            }
160            REF_NULL => {
161                stack.push_valtype(ValType::RefType(RefType::decode(wasm)?));
162            }
163            REF_FUNC => {
164                let func_idx = FuncIdx::decode_and_validate(wasm, c_funcs)?;
165
166                // This func_idx is automatically in C.refs. No need to check.
167                // as we are single pass validating, add it to C.refs set.
168                seen_func_idxs.push(func_idx);
169
170                stack.push_valtype(ValType::RefType(crate::RefType::FuncRef));
171            }
172            FD_EXTENSIONS => {
173                use crate::core::structure::instructions::fd_extensions::*;
174
175                let Ok(second_instr) = wasm.decode_var_u32() else {
176                    return Err(ValidationError::ExprMissingEnd);
177                };
178
179                trace!(
180                    "Validating const FD instruction {} at pc={}",
181                    fd_extension_instruction_to_str(second_instr),
182                    wasm.pc
183                );
184
185                match second_instr {
186                    V128_CONST => {
187                        for _ in 0..16 {
188                            let _data = wasm.decode_u8()?;
189                        }
190                        stack.push_valtype(ValType::VecType);
191                    }
192                    0x00..=0x0B | 0x0D.. => {
193                        trace!("Encountered unknown multi-byte instruction in validation - constant expression - {first_instr_byte:x?} {second_instr}");
194                        return Err(ValidationError::InvalidInstr(first_instr_byte));
195                    }
196                }
197            }
198
199            0x00..=0x0A
200            | 0x0C..=0x22
201            | 0x24..=0x40
202            | 0x45..=0xBF
203            | 0xC0..=0xCF
204            | 0xD1
205            | 0xD3..=0xFC
206            | 0xFE..=0xFF => {
207                trace!("Encountered unknown instruction in validation - constant expression - {first_instr_byte:x?}");
208                return Err(ValidationError::InvalidInstr(first_instr_byte));
209            }
210        }
211    }
