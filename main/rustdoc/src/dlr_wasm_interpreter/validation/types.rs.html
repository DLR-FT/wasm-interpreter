17impl BlockType {
18    /// Decodes a block type[^binary-format] and validates it[^validation].
19    ///
20    /// [^binary-format]: [WebAssembly Specification 2.0 - 5.4.1. Control Instructions](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#binary-blocktype).
21    /// [^validation]: [WebAssembly Specification 2.0 - 3.2.2. Block Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#valid-blocktype).
22    pub fn decode_and_validate(
23        wasm: &mut WasmDecoder,
24        c_types: &IdxVec<TypeIdx, FuncType>,
25    ) -> Result<Self, ValidationError> {
26        if wasm.peek_u8()? == 0x40 {
27            // Empty block type
28            let _ = wasm.decode_u8().unwrap_validated();
29            Ok(BlockType::Empty)
30        } else if let Ok(val_ty) = wasm.handle_transaction(|wasm| ValType::decode(wasm)) {
31            // No parameters and given valtype as the result
32            Ok(BlockType::Returns(val_ty))
33        } else {
34            // An index to a function type
35            let index = wasm.decode_var_i33_as_u32()?;
36            TypeIdx::validate(index, c_types).map(BlockType::Type)
37        }
38    }
39
40    /// Converts this block type to a specific [`FuncType`].
41    ///
42    /// A vector of function types is required, in case the current block type
43    /// stores a type index.
44    ///
45    /// # Safety
46    ///
47    /// The given [`IdxVec<TypeIdx, FuncType>`] must be the same on that was
48    /// used to validate `self` through [`BlockType::decode_and_validate`].
49    // TODO maybe make this function return a `Cow<'a, FuncType>`. This could
50    // prevent one allocation per call.
51    pub unsafe fn as_func_type(&self, func_types: &IdxVec<TypeIdx, FuncType>) -> FuncType {
52        match self {
53            BlockType::Empty => FuncType::new_empty(),
54            BlockType::Returns(val_type) => FuncType::new_returning(*val_type),
55            BlockType::Type(type_idx) => {
56                // SAFETY: The caller ensures that this `IdxVec` is the same one
57                // used to validate the `TypeIdx` in `self`.
58                unsafe { func_types.get(*type_idx) }.clone()
59            }
60        }
61    }
62}

64impl Limits {
65    /// Decodes a limits object[^binary-format] and validates it[^validation].
66    ///
67    /// [^binary-format]: [WebAssembly Specification 2.0 - 5.3.7. Limits](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#binary-limits).
68    /// [^validation]: [WebAssembly Specification 2.0 - 3.2.1. Limits](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#valid-limits).
69    pub fn decode_and_validate(
70        wasm: &mut WasmDecoder,
71        range: u32,
72    ) -> Result<Self, ValidationError> {
73        let limits = match wasm.decode_u8()? {
74            0x00 => {
75                let min = wasm.decode_var_u32()?;
76                Self { min, max: None }
77            }
78            0x01 => {
79                let min = wasm.decode_var_u32()?;
80                let max = wasm.decode_var_u32()?;
81                Self {
82                    min,
83                    max: Some(max),
84                }
85            }
86            other => return Err(DecodingError::MalformedLimitsDiscriminator(other).into()),
87        };
88
89        if limits.min > range {
90            return Err(ValidationError::LimitsNotWithinRange(range));
91        }
92
93        if let Some(max) = limits.max {
94            if max > range {
95                return Err(ValidationError::LimitsNotWithinRange(range));
96            }
97
98            if limits.min > max {
99                return Err(ValidationError::LimitsMinLargerThanMax {
100                    min: limits.min,
101                    max,
102                });
103            }
104        }
105
106        Ok(limits)
107    }
108}

110impl TableType {
111    /// Decodes a table type[^binary-format] and validates it[^validation].
112    ///
113    /// [^binary-format]: [WebAssembly Specification 2.0 - 5.3.9. Table Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#binary-tabletype).
114    /// [^validation]: [WebAssembly Specification 2.0 - 3.2.4. Table Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#valid-tabletype).
115    pub fn decode_and_validate(wasm: &mut WasmDecoder) -> Result<Self, ValidationError> {
116        const LIMITS_RANGE: u32 = u32::MAX; // = 2^32 - 1
117
118        let et = RefType::decode(wasm)?;
119        let lim = Limits::decode_and_validate(wasm, LIMITS_RANGE)?;
120
121        Ok(Self { et, lim })
122    }
123}

125impl MemType {
126    /// Decodes a memory type[^binary-format] and validates it[^validation].
127    ///
128    /// [^binary-format]: [WebAssembly Specification 2.0 - 5.3.8. Memory Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#binary-memtype).
129    /// [^validation]: [WebAssembly Specification 2.0 - 3.2.5. Memory Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#valid-memtype).
130    pub fn decode_and_validate(wasm: &mut WasmDecoder) -> Result<Self, ValidationError> {
131        const LIMITS_RANGE: u32 = 2_u32.pow(16);
132
133        let limits = Limits::decode_and_validate(wasm, LIMITS_RANGE)?;
134
135        Ok(Self { limits })
136    }
