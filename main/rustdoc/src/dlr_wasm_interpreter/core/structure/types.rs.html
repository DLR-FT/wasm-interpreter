1//! # Valid Type Definitions
2//!
3//! This module defines Rust structs for Wasm types as defined in [^structure-types], which haveh
4//! already been validated according to [^valid-types]. Some structs may always be valid, while
5//! other (e.g. [`Limits`]) have invariants associated with them which are not expressed through
6//! Wasm's type system.
7//!
8//! ## Types from the Instructions Chapter
9//!
10//! The Wasm specification defines most of its types in the types chapter[^structure-types].
11//! However, block types and memargs are defined in the instructions
12//! chapter[^structure-instructions] alongside the instructions which use them. For completeness
13//! these types are included in this Rust module as [`BlockType`] and [`MemArg`].
14//!
15//! [^structure-types]: [WebAssembly Specification 2.0 - 2.3. Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#types%E2%91%A0).
16//! [^valid-types]: [WebAssembly Specification 2.0 - 3.2. Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#types%E2%91%A4).
17//! [^structure-instructions]: [WebAssembly Specification 2.0 - 2.4. Instructions](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#instructions%E2%91%A0).

87/// A limits object[^spec]
88///
89/// This object is only valid within a certain range `k`[^validation] that depends on the context in
90/// which these limits exist. This can be a [`MemType`] or a [`TableType`].
91///
92/// [^spec]: [WebAssembly Specification 2.0 - 2.3.7. Limits](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#syntax-limits).
93/// [^validation]: [WebAssembly Specification 2.0 - 3.2.1. Limits](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#limits%E2%91%A2).
94#[derive(Copy, Clone, PartialEq, Eq)]
95pub struct Limits {
96    pub min: u32,
97    /// If this is `Some(n)`, `n` must be greater or equal to `self.min`
98    pub max: Option<u32>,
99}
100
101// TODO Find out where these consts come from and document it properly
102impl Limits {
103    // since the maximum amount of bytes is u32::MAX, the page size is 1 << 16
104    // the max no. of pages = max bytes / page size = u32::MAX / (1 << 16) = 1 << 16
105    pub const MAX_MEM_PAGES: u32 = 1 << 16;
106    // https://webassembly.github.io/reference-types/core/syntax/types.html#limits
107    // memtype is defined in terms of limits, which go from 0 to u32::MAX
108    pub const MAX_MEM_BYTES: u32 = u32::MAX;
109    // https://webassembly.github.io/reference-types/core/exec/runtime.html#memory-instances
110    // memory size is 65536 (1 << 16)
111    pub const MEM_PAGE_SIZE: u32 = 1 << 16;
112}
113
114impl fmt::Debug for Limits {
115    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> core::fmt::Result {
116        match self.max {
117            Some(max) => f.write_fmt(format_args!("{}..{}", self.min, max)),
118            None => f.write_fmt(format_args!("{}..", self.min)),
119        }
120    }
121}

123/// A table type
124///
125/// See: [WebAssembly Specification 2.0 - 2.3.9. Table Types](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#syntax-tabletype).

132    pub lim: Limits,
133}
