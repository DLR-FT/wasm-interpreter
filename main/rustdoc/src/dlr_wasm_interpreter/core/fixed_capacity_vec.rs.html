4/// The operation would remove more elements than currently present or tries to access an element
5/// when none are present
6#[derive(Debug, PartialEq, Eq, Clone)]

9/// The operation would add elements past the maximum capacity
10#[derive(Debug, PartialEq, Eq, Clone)]
11pub struct FullContainerError;

21pub struct FixedCapacityVec<T> {
22    /// A contiguous, non-reallocating, heap allocated vector. Behaves like a subset of
23    /// [`alloc::vec::Vec`], cleansed of any operation that reallocates. Backed by via boxed
24    /// slice containing elements of type `MaybeUninit<T>`. The maximum size (pendant to
25    /// [`alloc::vec::Vec::capacity`]) is determined at creation and remains unchanged over the
26    /// lifetime of [`Self`]. Nonetheless a variable amount of `T` can be kept within [`Self`], just
27    /// never more than the maximum size passed to the constructor.
28    ///
29    /// This can be used as a stack, in which case the first element (forming the bottom of the
30    /// stack) is at index 0, and new elements grow towards [`Self`]'s end. The last element
31    /// (`self.stack[self.stack.len() - 1]` is the topmost possible entry.
32    elements: Box<[MaybeUninit<T>]>,
33
34    /// Number of elements in [`Self`]. Also an index, pointing always to the first unused slot in
35    /// [`Self::elements`].
36    // Developer notes:
37    // - It is UB to access `self.elements[self.len]` or any other index larger than `self.len`.
38    // - If `self.stack_height == 0`, then there are no elements.
39    // - The guaranteed to be initialized range of [`Self::stack`] always is `self.stack[..self.stack_height]`.
40    len: usize,
41}
42
43// TODO Optimize Clone impl by using unsafe code
44impl<T: Clone> Clone for FixedCapacityVec<T> {
45    fn clone(&self) -> Self {
46        let mut cloned = Self::with_capacity(self.elements.len());
47
48        for index in 0..self.len {
49            let element = self
50                .get(index)
51                .expect("that this index is valid because it is less that self.len");
52            cloned
53                .push(element.clone())
54                .expect("that the new cloned vec has the same length as self and cannot overflow");
55        }
56
57        cloned
58    }
59}

61impl<T> FixedCapacityVec<T> {
62    /// Construct new [`Self`], holding up to `capacity` elements of type `T`
63    pub fn with_capacity(capacity: usize) -> Self {
64        Self {
65            elements: Box::new_uninit_slice(capacity),
66            len: 0,
67        }
68    }
69
70    /// Check if the [`Self`] is empty
71    #[inline(always)]
72    pub const fn is_empty(&self) -> bool {
73        self.len == 0
74    }
75
76    /// Check if the [`Self`] is full
77    #[inline(always)]
78    pub const fn is_full(&self) -> bool {
79        self.len == self.elements.len()
80    }
81
82    /// Get the [`Self`] is height.
83    #[inline(always)]
84    pub const fn len(&self) -> usize {
85        self.len
86    }
87
88    /// Get the maximum number of elements that fit into [`Self`].
89    #[inline(always)]
90    pub const fn capacity(&self) -> usize {
91        self.elements.len()
92    }
93
94    /// Push a value to the end of [`Self`]
95    ///
96    /// # Safety
97    ///
98    /// - Causes UB if [`Self`] is already full.
99    /// - Causes UB if [`Self`] has a capacity equal to `usize::MAX`
100    #[inline(always)]
101    pub unsafe fn push_unchecked(&mut self, value: T) {
102        debug_assert!(!self.is_full());
103        debug_assert!(self.capacity() < usize::MAX);
104
105        // SAFETY: This must never be called when `Self` is full
106        unsafe { *self.elements.get_unchecked_mut(self.len) = MaybeUninit::new(value) };
107
108        // SAFETY: This must never be called when `self.capacity()` is `usize::MAX`
109        self.len = unsafe { self.len.unchecked_add(1) };
110    }
111
112    /// Push a value to the end of [`Self`]
113    pub fn push(&mut self, value: T) -> Result<(), FullContainerError> {
114        // check if insertion will overflow `self.len`
115        let new_len = self.len.checked_add(1).ok_or(FullContainerError)?;
116
117        *self.elements.get_mut(self.len).ok_or(FullContainerError)? = MaybeUninit::new(value);
118
119        // only "commit" the `new_len` if the insertion actually succeeded
120        self.len = new_len;
121
122        Ok(())
123    }
124
125    /// Pop a value from [`Self`]'s top/tail
126    ///
127    /// # Safety
128    ///
129    /// Causes UB if the stack is empty.
130    #[inline(always)]
131    pub unsafe fn pop_unchecked(&mut self) -> T {
132        debug_assert!(!self.is_empty());
133
134        // SAFETY: This must never be called when `self` is empty. If the stack is indeed not
135        // empty, then both the unchecked subtraction is guaranteed not to underflow.
136        self.len = unsafe { self.len.unchecked_sub(1) };
137
138        // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the number
139        // of initialized values starting from `self.element`'s beginning. As we require `self.len`
140        // not to be zero before calling this function, it is now at zero and thus a valid index to
141        // an initialized element.
142        unsafe { self.elements.get_unchecked(self.len).assume_init_read() }
143    }
144
145    /// Pop a value from [`Self`]'s top/tail
146    pub fn pop(&mut self) -> Result<T, EmptyContainerError> {
147        self.len = self.len.checked_sub(1).ok_or(EmptyContainerError)?;
148
149        // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the number
150        // of initialized values starting from `self.element`'s beginning. As we require `self.len`
151        // not to be zero before calling this function, it is now at zero and thus a valid index to
152        // an initialized element.
153        let value = unsafe { self.elements.get_unchecked(self.len).assume_init_read() };
154
155        Ok(value)
156    }
157
158    /// Peek at the topmost/last value from [`Self`]
159    ///
160    /// # Safety
161    ///
162    /// Causes UB if the stack is empty.
163    #[inline(always)]
164    pub unsafe fn peek_unchecked(&self) -> &T {
165        debug_assert!(!self.is_empty());
166
167        // SAFETY: This must never be called when the `self` is empty. If the stack is indeed not
168        // empty, then both the unchecked subtraction can not underflow, and the element at that
169        // index will be already initialized.
170        unsafe {
171            self.elements
172                .get_unchecked(self.len.unchecked_sub(1))
173                .assume_init_ref()
174        }
175    }
176
177    /// Peek at the topmost/last value from [`Self`]
178    pub fn peek(&self) -> Result<&T, EmptyContainerError> {
179        let idx = self.len.checked_sub(1).ok_or(EmptyContainerError)?;
180
181        // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the number
182        // of initialized values starting from `self.element`'s beginning. As we require `self.len`
183        // not to be zero before calling this function, it is now at zero and thus a valid index to
184        // an initialized element.
185        let value = unsafe { self.elements.get_unchecked(idx).assume_init_ref() };
186
187        Ok(value)
188    }
189
190    /// Get a shared ref to the nth element in [`Self`]
191    #[inline(always)]
192    pub fn get(&self, idx: usize) -> Option<&T> {
193        (idx < self.len).then(|| {
194            // SAFETY: `idx` is less than `self.len()`
195            unsafe { self.get_unchecked(idx) }
196        })
197    }
198
199    /// Get a shared ref to the nth element in [`Self`]
200    ///
201    /// # Safety
202    ///
203    /// `idx` must be within bounds, i.e. `0 <= idx < self.len()`.
204    #[inline(always)]
205    pub unsafe fn get_unchecked(&self, idx: usize) -> &T {
206        debug_assert!(self.elements.get(idx).is_some());
207        // SAFETY: The caller ensures that `idx` is less than `self.len()` which it always less or
208        // equal to `self.capacity()`. Therefore, `idx` is always less than `self.capacity()`.
209        let element = unsafe { self.elements.get_unchecked(idx) };
210
211        debug_assert!(idx < self.len());
212        // SAFETY: The caller ensures that `idx` is less than `self.len()`. Therefore, the element
213        // at this index must be initialized.
214        unsafe { element.assume_init_ref() }
215    }
216
217    /// Get an exclusive reference to the nth element in [`Self`]
218    #[inline(always)]
219    #[expect(unused, reason = "this might be used in the future")]
220    pub fn get_mut(&mut self, idx: usize) -> Option<&mut T> {
221        (idx < self.len).then(|| {
222            // SAFETY: `idx` is less than `self.len()`
223            unsafe { self.get_unchecked_mut(idx) }
224        })
225    }
226
227    /// Get an exclusive reference to the n-th element in [`Self`]
228    ///
229    /// # Safety
230    ///
231    /// `idx` must be within bounds, i.e. `0 <= idx < self.len()`.
232    #[inline(always)]
233    pub unsafe fn get_unchecked_mut(&mut self, idx: usize) -> &mut T {
234        debug_assert!(self.elements.get(idx).is_some());
235        debug_assert!(idx < self.len());
236
237        // SAFETY: The caller ensures that `idx` is less than `self.len()` which it always less or
238        // equal to `self.capacity()`. Therefore, `idx` is always less than `self.capacity()`.
239        let element = unsafe { self.elements.get_unchecked_mut(idx) };
240
241        // SAFETY: The caller ensures that `idx` is less than `self.len()`. Therefore, the element
242        // at this index is guaranteed to be initialized.
243        unsafe { element.assume_init_mut() }
244    }
245}

247impl<T: Clone> FixedCapacityVec<T> {
248    /// Push from a slice into [`Self`], appending after the last/topmost element
249    pub fn push_from_slice(&mut self, values: &[T]) -> Result<(), FullContainerError> {
250        // verify `values` fits into the new self
251        if values.len() > (self.capacity() - self.len) {
252            return Err(FullContainerError);
253        }
254
255        for e in values {
256            // SAFETY: this is safe, as the previous if statement verifies that there is enough
257            // space for all elements from `values` in `stack`
258            unsafe { self.push_unchecked(e.clone()) };
259        }
260
261        Ok(())
262    }
263
264    /// Pop `n` elements from [`Self`] into a slice. The topmost/last element of [`Self`] will
265    /// become the slice's last element.
266    pub fn pop_into_slice(
267        &mut self,
268        n: usize,
269    ) -> Result<impl core::ops::Deref<Target = [T]> + '_, EmptyContainerError> {
270        // verify at least `n` elements are still one the stack
271        if n > self.len() {
272            return Err(EmptyContainerError);
273        }
274
275        let old_len = self.len();
276
277        // SAFETY: the previous if statement ensure that `n` is smaller than or equal to
278        // `self.len()`, hence this subtraction is guaranteed to not underflow
279        self.len = unsafe { self.len.unchecked_sub(n) };
280
281        // SAFETY: the prior if statement checks that `stack_height - n` wont underflow. And the
282        // rest of the module ensures that stack_height never gets bigger than `self.stack.len()`.
283        // Thus the index range is always in-bounds.
284        Ok(SliceDropGuard(unsafe {
285            self.elements.get_unchecked_mut(self.len..old_len)
286        }))
287    }
288}

290pub struct SliceDropGuard<'a, T>(&'a mut [MaybeUninit<T>]);
291
292impl<'a, T> core::ops::Deref for SliceDropGuard<'a, T> {
293    type Target = [T];
294
295    fn deref(&self) -> &Self::Target {
296        // SAFETY: The rest of the module ensures that stack_height never gets bigger than
297        // `self.stack.len()`. Thus the index range is always in-bounds.
298        unsafe { slice_assume_init(self.0) }
299    }
300}
301
302impl<'a, T> Drop for SliceDropGuard<'a, T> {
303    fn drop(&mut self) {
304        for e in &mut *self.0 {
305            // SAFETY: The rest of the module ensures that stack_height never gets bigger than
306            // `self.stack.len()`. Thus the index range is always in-bounds.
307            unsafe { e.assume_init_drop() };
308        }
309    }
310}

312impl<T: Copy> FixedCapacityVec<T> {
313    /// Remove `remove_count` values from [`Self`], keeping the topmost `keep_count` values
314    ///
315    /// From the [`Self`], remove `remove_count` elements, by sliding down the `keep_count` last/topmost
316    /// values `remove_count` positions forward/towards the bottom.
317    ///
318    /// **Effects**
319    ///
320    /// - after the operation, [`Self`] will contain `remove_count` fewer elements
321    /// - `keep_count` topmost elements will be identical before and after the operation
322    /// - all elements below the `remove_count + keep_count` topmost stack entry remain
323    pub fn remove_in_between(&mut self, remove_count: usize, keep_count: usize) {
324        // TODO make unchecked version, remove overflowing arithmetic in safe version
325        let len = self.len();
326        self.elements
327            .copy_within(len - keep_count..len, len - keep_count - remove_count);
328        self.len -= remove_count;
329    }
330}

332impl<T: core::fmt::Debug> core::fmt::Debug for FixedCapacityVec<T> {
333    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
334        // SAFETY: per the guarantees of push/pop/etc., `0..self.stack_height` is always a range to
335        // initalized values.
336        let stack_elements_slice = unsafe { slice_assume_init(&self.elements[..self.len]) };
337
338        f.debug_struct("Stack")
339            .field("stack", &stack_elements_slice)
340            .field("stack_height", &self.len)
341            .finish()
342    }
343}

345impl<T> Drop for FixedCapacityVec<T> {
346    fn drop(&mut self) {
347        // SAFETY: self is guaranteed to have at least `self.len` elements, because `sel.len` can
348        // not grow past `self.elements.len` (aka `self.capacity()`).
349        for e in unsafe { self.elements.get_unchecked_mut(0..self.len) } {
350            // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the
351            // number of initialized values starting from `self.element`'s beginning.
352            unsafe { e.assume_init_drop() }
353        }
354    }

357// TODO use assume_init_ref, once we get the MSRV to 1.93.0
358/// # Safety
359///
360/// Use this to assume a range of a slice to be initialized. This is provided starting with Rust
361/// 1.93.0.
362#[inline(always)]
363unsafe fn slice_assume_init<T>(slice: &[MaybeUninit<T>]) -> &[T] {
364    // SAFETY: casting `slice` to a `*const [T]` is safe since the caller guarantees that
365    // `slice` is initialized, and `MaybeUninit` is guaranteed to have the same layout as `T`.
366    // The pointer obtained is valid since it refers to memory owned by `slice` which is a
367    // reference and thus guaranteed to be valid for reads.
368    unsafe { &*(slice as *const _ as *const [T]) }
369}
