1//! An in-place interpreter for WebAssembly 2.0
2//!
3//! # General Usage
4//!
5//! WebAssembly (Wasm) modules must first be decoded and validated through [`decode_and_validate`],
6//! producing a [`Module`]. This module can then be instantiated in a [`Store`] via
7//! [`Store::module_instantiate`], creating a module instance and returning its module address,
8//! uniquely identifying this module instance within that store.
9//!
10//! When a [`Store`] is initially created through [`Store::new`], it is empty. This store exposes
11//! many other functions besides module instantiation to interact with it and objects allocated
12//! within it. Most notably, the [`Store::invoke_simple`] and [`Store::invoke`] methods are used to
13//! interpret Wasm code. Refer to the examples in `examples/` for more information.
14//!
15//! # Example
16//!
17//! This is an example for how to run a function exposed from a simple Wasm module (full code in
18//! `examples/function_invocation.rs`):
19//!
20//! ```
21//! # use dlr_wasm_interpreter::{ExternVal, FuncAddr, InstantiationOutcome, Module, Store, Value, decode_and_validate};
22//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
23//! const WAT_CODE: &str = r#"
24//! (module
25//!     (func (export "add_one") (param $n i32) (result i32)
26//!         local.get $n
27//!         i32.const 1
28//!         i32.add))
29//! "#;
30//!
31//! // Use the `wat` crate to convert from the text format to bytecode
32//! let wasm_bytecode = wat::parse_str(WAT_CODE)?;
33//!
34//! // Decode and validate the module
35//! let module = decode_and_validate(&wasm_bytecode, &mut ())?;
36//!
37//! // Create a new empty store
38//! let mut store = Store::new(());
39//!
40//! // Instantiate the module to create a module instance, returning its address
41//! // SAFETY: There are no extern values.
42//! let module_addr = unsafe { store.module_instantiate(&module, vec![], None) }?.module_addr;
43//!
44//! // Get the function address of the exported add_one function
45//! // SAFETY: The module address was returned from the same store.
46//! let add_one_extern = unsafe { store.instance_export(module_addr, "add_one") }?;
47//! let add_one = add_one_extern.as_func().ok_or("add_one is not a function")?;
48//!
49//! // Invoke the function
50//! // SAFETY: The function address was returned from the same store. There are also no address
51//! // type parameters.
52//! let return_values = unsafe { store.invoke_simple(add_one, vec![Value::I32(16)]) }?;
53//! assert_eq!(*return_values, [Value::I32(17)]);
54//! # Ok(())
55//! # }
56//! ```

90/// A definition for a [`Result`] using the optional [`Error`] type.
91pub type Result<T> = ::core::result::Result<T, Error>;
92
93/// An opt-in error type useful for merging all error types of this crate into a single type.
94///
95/// Note: This crate does not use this type in any public interfaces, making it optional for downstream users.
96#[derive(Debug, PartialEq, Eq)]
97pub enum Error {
98    Validation(ValidationError),
99    RuntimeError(RuntimeError),
100}
101
102impl From<ValidationError> for Error {
103    fn from(value: ValidationError) -> Self {
104        Self::Validation(value)
105    }
106}
107
108impl From<RuntimeError> for Error {
109    fn from(value: RuntimeError) -> Self {
110        Self::RuntimeError(value)
111    }
112}

124            error,
125            Error::Validation(ValidationError::Decoding(DecodingError::InvalidMagic))
126        )
127    }
128
129    #[test]

131        let runtime_error = RuntimeError::ModuleNotFound;
132        let error: Error = runtime_error.into();
133
134        assert_eq!(error, Error::RuntimeError(RuntimeError::ModuleNotFound))
135    }
