7/// A linear memory is the backing data structure for a memory instance[^memory-instances].
8///
9/// It is a wrapper around a contiguous, but growable sequence of bytes. Also it provides generic
10/// methods to be build upon by memory instructions[^memory-instructions].
11///
12/// TODO: Write section on why some memory instructions are implemented here vs. implemented in
13/// their instruction handlers. Also find out how we can reference the specification steps for each
14/// instruction.
15///
16///
17/// [^memory-instances]: [WebAssembly Specification 2.0 - 4.2.9. Memory Instances](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#memory-instances%E2%91%A0).
18/// [^memory-instructions]: [WebAssembly Specification 2.0 - 4.4.7. Memory Instructions](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#memory-instructions%E2%91%A4).
19pub struct LinearMemory<const PAGE_SIZE: usize = { crate::Limits::MEM_PAGE_SIZE as usize }> {
20    pub(crate) data: Vec<u8>,
21}
22
23/// Type to express the page count
24pub type PageCountTy = u16;
25
26impl<const PAGE_SIZE: usize> LinearMemory<PAGE_SIZE> {
27    /// Creates a new linear memory of size 0
28    pub fn new() -> Self {
29        Self { data: Vec::new() }
30    }
31
32    /// Creates a new zero-initialized linear memory. Its size is determined by the given number of
33    /// pages and the `PAGE_SIZE`.
34    pub fn new_with_initial_pages(pages: PageCountTy) -> Self {
35        let size_bytes = PAGE_SIZE * usize::from(pages);
36        let data = vec![0; size_bytes];
37
38        Self { data }
39    }
40
41    /// Grows the current linear memory by a number of pages.
42    pub fn grow(&mut self, pages_to_add: PageCountTy) {
43        let prior_length_bytes = self.data.len();
44        let new_length_bytes = prior_length_bytes + PAGE_SIZE * usize::from(pages_to_add);
45        self.data.resize(new_length_bytes, 0);
46    }
47
48    /// Returns the size of this linear memory in pages.
49    pub fn pages(&self) -> PageCountTy {
50        PageCountTy::try_from(self.data.len() / PAGE_SIZE).unwrap()
51    }
52
53    /// Returns the size of this linear memory in bytes.
54    ///
55    /// This size is always a multiple of `PAGE_SIZE`.
56    pub fn len(&self) -> usize {
57        self.data.len()
58    }
59
60    /// Stores a `T` starting from the given index into this linear memory.
61    ///
62    /// The `T` must be convertible to its little-endian byte form (as required by the
63    /// [`LittleEndianBytes`] bound).
64    ///
65    /// # Errors
66    ///
67    /// Same as [`LinearMemory::store_bytes`]
68    pub fn store<const N: usize, T: LittleEndianBytes<N>>(
69        &mut self,
70        index: usize,
71        value: T,
72    ) -> Result<(), TrapError> {
73        self.store_bytes::<N>(index, value.to_le_bytes())
74    }
75
76    /// Stores a number of bytes into this linear memory starting at the given index.
77    ///
78    /// # Errors
79    ///
80    /// - [`TrapError::MemoryOrDataAccessOutOfBounds`]: The store would have been out of bounds. The
81    ///   memory remains unchanged.
82    pub fn store_bytes<const N: usize>(
83        &mut self,
84        index: usize,
85        bytes: [u8; N],
86    ) -> Result<(), TrapError> {
87        let target_range = index..(index + N);
88
89        let target_bytes = self
90            .data
91            .get_mut(target_range)
92            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
93
94        target_bytes.copy_from_slice(&bytes);
95
96        Ok(())
97    }
98
99    /// Loads a `T` starting from the given index into this linear memory.
100    ///
101    /// The `T` must be convertible from its little-endian byte form (as required by the
102    /// [`LittleEndianBytes`] bound).
103    ///
104    /// # Errors
105    ///
106    /// Same as [`LinearMemory::load_bytes`]
107    pub fn load<const N: usize, T: LittleEndianBytes<N>>(
108        &self,
109        index: usize,
110    ) -> Result<T, TrapError> {
111        self.load_bytes::<N>(index).map(T::from_le_bytes)
112    }
113
114    /// Loads a number of bytes from this linear memory starting at the given index.
115    ///
116    /// # Errors
117    ///
118    /// - [`TrapError::MemoryOrDataAccessOutOfBounds`]: The load would have been out of bounds.
119    pub fn load_bytes<const N: usize>(&self, index: usize) -> Result<[u8; N], TrapError> {
120        let target_range = index..(index + N);
121
122        let target_bytes = self
123            .data
124            .get(target_range)
125            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
126
127        let bytes: [u8; N] = target_bytes
128            .try_into()
129            .expect("bytes slice to be of length N");
130
131        Ok(bytes)
132    }
133
134    /// Sets a number of bytes in this linear memory to a constant `data_byte`. A total of `count`
135    /// bytes are written, starting at the given index.
136    ///
137    /// # Errors
138    ///
139    /// - [`TrapError::MemoryOrDataAccessOutOfBounds`]: The fill operation would have been out of
140    ///   bounds. The memory remains unchanged.
141    pub fn fill(&mut self, index: usize, data_byte: u8, count: usize) -> Result<(), TrapError> {
142        let target_range = index..(index + count);
143
144        /* check destination for out of bounds access */
145        // Specification step 12.
146        let target_bytes = self
147            .data
148            .get_mut(target_range)
149            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
150
151        /* do the fill, do nothing if count was zero */
152        // Specification step 13-21.
153        target_bytes.fill(data_byte);
154
155        Ok(())
156    }
157
158    /// Copies a number of bytes inside this linear memory from one to another location. Both the
159    /// source and destination may overlap.
160    ///
161    /// A total of `count` bytes will be copied, starting at the given source and destination
162    /// indices.
163    ///
164    /// # Errors
165    ///
166    /// - [`TrapError::MemoryOrDataAccessOutOfBounds`]: The copy operation would have been out of
167    ///   bounds. The memory remains unchanged.
168    pub fn copy_within(
169        &mut self,
170        destination_index: usize,
171        source_index: usize,
172        count: usize,
173    ) -> Result<(), RuntimeError> {
174        let source_range = source_index..(source_index + count);
175        let destination_range = destination_index..(destination_index + count);
176
177        /* check source and destination for out of bounds accesses */
178        // Specification step 12.
179        let source_range_is_within_bounds = self.data.get(source_range.clone()).is_some();
180        let destination_range_is_within_bounds = self.data.get(destination_range).is_some();
181        if !source_range_is_within_bounds || !destination_range_is_within_bounds {
182            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
183        }
184
185        /* do the potentially-overlapping copy, does nothing if count was zero */
186        // Specification step 13-15
187
188        // This cannot panic, because both ranges are within bounds.
189        self.data.copy_within(source_range, destination_index);
190
191        Ok(())
192    }
193
194    /// Copies a number of bytes from the given byte slice `source_data` into this linear memory. A
195    /// total of `count` bytes will be copied, starting at the given source and destination indices.
196    ///
197    /// # Errors
198    ///
199    /// - [`TrapError::MemoryOrDataAccessOutOfBounds`]: This operation would have been out of
200    ///   bounds. The memory remains unchanged.
201    pub fn init(
202        &mut self,
203        destination_index: usize,
204        source_data: &[u8],
205        source_index: usize,
206        count: usize,
207    ) -> Result<(), RuntimeError> {
208        let source_range = source_index..(source_index + count);
209        let destination_range = destination_index..(destination_index + count);
210
211        /* check source and destination for out of bounds accesses */
212        // Specification step 16.
213        let source_bytes = source_data
214            .get(source_range)
215            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
216        let destination_bytes = self
217            .data
218            .get_mut(destination_range)
219            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
220
221        /* do the init, also check if there is anything to be done */
222        // Specification step 17-27.
223        destination_bytes.copy_from_slice(source_bytes);
224
225        Ok(())
226    }
227
228    /// Returns the data in this memory as a byte slice. The length of this slice is always a
229    /// multiple of `PAGE_SIZE`.
230    pub fn data_mut(&mut self) -> &mut [u8] {
231        &mut self.data
232    }
233}

235impl<const PAGE_SIZE: usize> core::fmt::Debug for LinearMemory<PAGE_SIZE> {
236    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
237        /// A helper struct for formatting, able to detect and format byte repetitions in a compact
238        /// way.
239        struct RepetitionDetectingMemoryWriter<'a>(&'a [u8]);
240        impl core::fmt::Debug for RepetitionDetectingMemoryWriter<'_> {
241            fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
242                /// The number of repetitions required for successive elements to be grouped
243                // together.
244                const MIN_REPETITIONS_FOR_GROUP: usize = 8;
245
246                // First we create an iterator over all bytes
247                let mut bytes = self.0.iter();
248
249                // Then we iterate over all bytes and deduplicate repetitions. This produces an
250                // iterator of pairs, consisting of the number of repetitions and the repeated byte
251                // itself. `current_group` is captured by the iterator and used as state to track
252                // the current group.
253                let mut current_group: Option<(usize, u8)> = None;
254                let deduplicated_with_count = iter::from_fn(|| {
255                    for &byte in bytes.by_ref() {
256                        // If the next byte is different than the one being tracked currently...
257                        if current_group.is_some() && current_group.unwrap().1 != byte {
258                            // ...then end and emit the current group but also start a new group for
259                            // the next byte with an initial count of 1.
260                            return current_group.replace((1, byte));
261                        }
262                        // Otherwise increment the current group's counter or start a new group if
263                        // this was the first byte.
264                        current_group.get_or_insert((0, byte)).0 += 1;
265                    }
266                    // In the end when there are no more bytes to read, directly emit the last
267                    current_group.take()
268                });
269
270                // Finally we use `DebugList` to print a list of all groups, while writing out all
271                // elements from groups with less than `MIN_REPETITIONS_FOR_GROUP` elements.
272                let mut list = f.debug_list();
273                deduplicated_with_count.for_each(|(count, value)| {
274                    if count < MIN_REPETITIONS_FOR_GROUP {
275                        list.entries(iter::repeat_n(value, count));
276                    } else {
277                        list.entry(&format_args!("#{count} × {value}"));
278                    }
279                });
280                list.finish()
281            }
282        }
283
284        // Format the linear memory by using Rust's formatter helpers and the previously defined
285        // `RepetitionDetectingMemoryWriter`
286        f.debug_struct("LinearMemory")
287            .field("inner_data", &RepetitionDetectingMemoryWriter(&self.data))
288            .finish()
289    }
290}
