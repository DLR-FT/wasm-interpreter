19/// Implementation of a shared linear memory, as defined in the threads proposal[^threads-proposal].
20///
21/// Unlike unshared linear memories which store their data as a vector of bytes, a shared linear
22/// memory's backing data structure is not specified.
23///
24/// Instead, a number of so-called _actions_, which are basically the side-effects of certain
25/// instructions, are defined. The following three types of actions are implemented as of now:
26///
27/// - `rd`: Reads the length or data of the memory. ([`Self::rd_len`], [`Self::rd_data`])
28/// - `wr`: Writes data into the memory. ([`Self::wr_data`])
29///
30/// These actions act as a base for implementing the execution of memory instructions as described
31/// in the threads proposal[^threads-proposal].
32///
33/// [^threads-proposal]: [Threads Proposal Repository](https://github.com/WebAssembly/threads).
34///
35/// # Implementation & Locking
36///
37/// This memory internally relies on a [`Vec<AtomicU8>`]. Thus, the atomic unit of information for
38/// it is a byte (`u8`). All byte-wise accesses to the linear memory internally occur through
39/// [`AtomicU8::load`] and [`AtomicU8::store`], avoiding the need for an exclusive for these
40/// operations.
41///
42/// However, there a two reasons for why a lock must guard the inner vector of atomic bytes:
43///
44/// 1. Implementing atomic stores for multi-byte values requires a global write lock. Rust's memory
45///    model considers partially overlapping atomic operations involving a write to be undefined
46///    behavior. It is also impossible to predict whether an atomic multi-byte store operation might
47///    overlap with another operation at runtime, thereby necessitating a lock to gain temporary
48///    exclusive access.
49/// 2. Linear memory can grow and while shared memories must always have an upper limit, we choose
50///    not to pre-allocate the entire memory. Instead we allow growing the inner [`Vec`] allocation,
51///    which requires temporary exclusive access to move all data from the old to the new fresh
52///    allocation. The old allocation is then freed afterwards.
53///
54/// TODO: Does it pay of to have more fine-granular locking for multi-byte stores than a single
55///       global write lock? Would this cause issues with predictable execution times?
56///
57/// # Unsafe Note
58///
59/// As the manual index checking assures all indices to be valid, there is no need to re-check.
60/// Therefore [`slice::get_unchecked`] is used access the internal [`AtomicU8`] in the vector
61/// backing a [`SharedLinearMemory`], implicating the use of `unsafe`.
62///
63/// To gain some confidence in the correctness of the unsafe code in this module, run `miri`:
64///
65/// ```bash
66/// cargo miri test --test memory # quick
67/// cargo miri test # thorough
68/// ```
69///
70// TODO: Allow rd/wr actions to take values directly instead of byte slices
71// TODO: Make user-defined limits (T::MAX_NUMBER_OF_MEMORY_PAGES) use the max_pages field, when we
72//       implement dynamically checked custom memory limits.
73pub struct SharedLinearMemory {
74    /// This vector's length must never be 2^32 or larger.
75    inner_data: RwSpinLock<Vec<AtomicU8>>,
76    page_size: NonZeroUsize,
77    /// We have to store the maximum number of pages in this memory so that we can perform a bounds
78    /// when whenever the user calls [`SharedLinearMemory::grow`].
79    max_pages: usize,
80}

82/// A memory ordering used for atomic accesses
83///
84/// See: [WebAssembly Specification 2.0 (with Threads proposal) - 4.2.17 Events - ord](https://webassembly.github.io/threads/core/exec/runtime.html#events)
85#[derive(Copy, Clone, Debug, PartialEq, Eq)]
86pub enum Ordering {
87    Unord,
88    SeqCst,
89    Init,
90}

121    // This is pub(crate) because we cannot expose custom page sizes to users yet.
122    // TODO For the custom page size proposal, rename this to `new_with_initial_pages`.
123    pub(crate) fn new_with_initial_pages_and_page_size<T: Config>(
124        pages: usize,
125        max_pages: usize,
126        page_size: NonZeroUsize,
127    ) -> Result<Self, RuntimeError> {
128        let size_bytes = page_size
129            .get()
130            .checked_mul(pages)
131            .ok_or(RuntimeError::MemoryOverflowed)?;
132
133        if let Some(max_pages) = T::MAX_NUMBER_OF_MEMORY_PAGES {
134            if pages > usize::from(max_pages.get()) {
135                // TODO Should we return RuntimeError::MemoryExceededLimit instead? Do this after we
136                // have proper unverified/verified versions of Limits/MemType (see issue #406).
137                return Err(RuntimeError::MemoryOverflowed);
138            }
139        }
140
141        let mut data = Vec::with_capacity(size_bytes);
142        data.resize_with(size_bytes, || AtomicU8::new(0));
143
144        Ok(Self {
145            inner_data: RwSpinLock::new(data),
146            page_size,
147            max_pages,
148        })
149    }

161    /// Grow the [`SharedLinearMemory`] by a number of pages, and fills the new bytes with zeros.
162    ///
163    /// The entire grow operation is atomic and visible to other threads as if it was
164    /// sequentially-consistent. Internally it works by temporarily acquiring exclusive access to
165    /// the memory's data storage.
166    pub fn grow<T: Config>(&self, pages_to_add: usize) -> Result<usize, RuntimeError> {
167        let mut lock_guard = self.inner_data.write();
168
169        let prior_length_bytes = lock_guard.len();
170        let len_pages = prior_length_bytes / self.page_size;
171        if len_pages
172            .checked_add(pages_to_add)
173            .is_none_or(|new_pages| new_pages > self.max_pages)
174        {
175            return Err(RuntimeError::MemoryGrowExceededLimit);
176        }
177
178        let num_bytes_to_append = self
179            .page_size
180            .get()
181            .checked_mul(pages_to_add)
182            .ok_or(RuntimeError::MemoryOverflowed)?;
183        let new_length_bytes = prior_length_bytes
184            .checked_add(num_bytes_to_append)
185            .ok_or(RuntimeError::MemoryOverflowed)?;
186
187        if let Some(max_pages) = T::MAX_NUMBER_OF_MEMORY_PAGES {
188            if usize::from(max_pages.get())
189                .checked_mul(self.page_size.get())
190                .is_some_and(|max_len| new_length_bytes > max_len)
191            {
192                return Err(RuntimeError::MemoryOverflowed);
193            }
194        }
195
196        lock_guard.resize_with(new_length_bytes, || AtomicU8::new(0));
197
198        Ok(len_pages)
199    }

209    /// At a given index, store a datum in the [`SharedLinearMemory`]
210    pub fn store<const N: usize, T: LittleEndianBytes<N>>(
211        &self,
212        index: usize,
213        value: T,
214        ord: Ordering,
215    ) -> Result<(), RuntimeError> {
216        // 13.
217        let index_is_properly_aligned = index.is_multiple_of(N);
218        if ord == Ordering::SeqCst && !index_is_properly_aligned {
219            return Err(TrapError::UnalignedAtomicAccess.into());
220        }
221
222        // 16.
223        let bytes = value.to_le_bytes();
224        // 18. The current memory is shared
225        // a.
226        let n = self.rd_len(Ordering::Unord);
227        // b.
228        let end_index_is_out_of_bounds = index.checked_add(N).is_none_or(|end_index| end_index > n);
229        if end_index_is_out_of_bounds {
230            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
231        }
232        // c.
233        // TODO There is a contradiction: tearing is allowed for types with <= 32 bits. However, 64
234        // bit atomic accesses could still be required to not tear as described here:
235        // https://github.com/WebAssembly/threads/issues/9#issuecomment-725389629
236        let may_tear: bool = tearing::<N, T>(index);
237
238        // d.
239        // SAFETY: The end index addition did not overflow and it is not larger than the length `n`
240        // that as just read.
241        unsafe { self.wr_data(index, bytes.into_iter(), ord, may_tear) };
242
243        Ok(())
244    }

246    /// At a given index, store a datum in the [`SharedLinearMemory`]
247    ///
248    /// This operation may tear.
249    pub fn store_bytes<const N: usize>(
250        &self,
251        index: usize,
252        bytes: [u8; N],
253        ord: Ordering,
254    ) -> Result<(), RuntimeError> {
255        let end_index = index
256            .checked_add(N)
257            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
258
259        let len = self.rd_len(Ordering::Unord);
260        if end_index > len {
261            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
262        }
263
264        // SAFETY: calculation of the the end index does not overflow and the end index is not
265        // larger than the length that was just read.
266        unsafe { self.wr_data(index, bytes.into_iter(), ord, true) };
267
268        Ok(())
269    }

271    /// From a given index, load a datum from the [`SharedLinearMemory`]
272    pub fn load<const N: usize, T: LittleEndianBytes<N>>(
273        &self,
274        index: usize,
275        ord: Ordering,
276    ) -> Result<T, RuntimeError> {
277        // 13.
278        let index_is_properly_aligned = index.is_multiple_of(N);
279        if ord == Ordering::SeqCst && !index_is_properly_aligned {
280            return Err(TrapError::UnalignedAtomicAccess.into());
281        }
282
283        // 18. The current memory is shared
284        // a.
285        let n = self.rd_len(Ordering::Unord);
286        // b.
287        let end_index_is_out_of_bounds = index.checked_add(N).is_none_or(|end_index| end_index > n);
288        if end_index_is_out_of_bounds {
289            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
290        }
291        // c.
292        // TODO There is a contradiction: tearing is allowed for types with <= 32 bits. However, 64
293        // bit atomic accesses could still be required to not tear as described here:
294        // https://github.com/WebAssembly/threads/issues/9#issuecomment-725389629
295        let may_tear: bool = tearing::<N, T>(index);
296
297        // d.
298        // SAFETY: The end index addition does not overflow and it is not larger than the length `n`
299        // that was just read.
300        let bytes = unsafe { self.rd_data_into_array::<N>(index, ord, may_tear) };
301
302        Ok(T::from_le_bytes(bytes))
303    }

325    /// The `memory.fill` instruction. It sets a number of bytes `count` in this memory to a
326    /// specific constant `data_byte`, starting at index `index`.
327    ///
328    /// Note that the Wasm specification defines this instruction by recursion, while this
329    /// implementation uses iteration.
330    ///
331    /// # See also
332    ///
333    /// - [WebAssembly Specification 2.0 (with Threads Proposal) - 4.4.7 Memory Instructions - mem.fill](https://webassembly.github.io/threads/core/exec/instructions.html#xref-syntax-instructions-syntax-instr-memory-mathsf-memory-fill)
334    pub fn fill(&self, index: usize, data_byte: u8, count: usize) -> Result<(), RuntimeError> {
335        let end_index = index
336            .checked_add(count)
337            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
338
339        let len = self.rd_len(Ordering::Unord);
340
341        if end_index > len {
342            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
343        }
344
345        let source_buffer_iterator = iter::repeat_n(data_byte, count);
346
347        // SAFETY: The end index did not overflow and is not larger than the length that was just
348        // read.
349        unsafe { self.wr_data_tearing(index, source_buffer_iterator, Ordering::Unord) };
350
351        Ok(())
352    }

354    /// The `memory.copy` instruction. It copies a number of bytes `count` from one region in this
355    /// memory to another region.  Starting indices must be specified for both the source and
356    /// destination regions, which may also overlap.
357    ///
358    /// Note that the reference specification for this implementation does not support multiple
359    /// memories. Therefore, no operation to copy data between multiple memories exists.
360    ///
361    /// # See also
362    ///
363    /// - [WebAssembly Specification 2.0 (with Threads Proposal) - 4.4.7 Memory Instructions - mem.copy](https://webassembly.github.io/threads/core/exec/instructions.html#xref-syntax-instructions-syntax-instr-memory-mathsf-memory-copy)
364    pub fn copy_within(
365        &self,
366        destination_index: usize,
367        source_index: usize,
368        count: usize,
369    ) -> Result<(), RuntimeError> {
370        let source_end_index = source_index
371            .checked_add(count)
372            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
373        let destination_end_index = destination_index
374            .checked_add(count)
375            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
376
377        let len = self.rd_len(Ordering::Unord);
378
379        if source_end_index > len || destination_end_index > len {
380            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
381        }
382
383        // We have to perform the copy backwards if certain overlaps are possible
384        if destination_index < source_index {
385            // Forwards
386            for offset in 0..count {
387                debug_assert!(source_index.checked_add(offset).is_some());
388                // SAFETY: The index addition producing the end index does not overflow, it is
389                // `source_end_index`.`offset` is also strictly smaller than count, therefore this
390                // cannot not overflow.
391                let final_source_index = unsafe { source_index.unchecked_add(offset) };
392
393                debug_assert!(destination_index.checked_add(offset).is_some());
394                // SAFETY: The index addition producing the end index does not overflow, it is
395                // `destination_end_index`. `offset` is also strictly smaller than count, therefore
396                // this cannot not overflow.
397                let final_destination_index = unsafe { destination_index.unchecked_add(offset) };
398
399                debug_assert!(final_source_index < len);
400                debug_assert!(final_destination_index < len);
401                // SAFETY: Both the final source and destination indices must be smaller than
402                // source_end_index and source_destination index which are smaller than len,
403                // respectively.
404                unsafe {
405                    self.copy_byte(final_source_index, final_destination_index, Ordering::Unord)
406                };
407            }
408        } else {
409            // Backwards
410            for offset in (0..count).rev() {
411                debug_assert!(source_index.checked_add(offset).is_some());
412                // SAFETY: The index addition producing the end index does not overflow, it is
413                // `source_end_index`.`offset` is also strictly smaller than count, therefore this
414                // cannot not overflow.
415                let final_source_index = unsafe { source_index.unchecked_add(offset) };
416
417                debug_assert!(destination_index.checked_add(offset).is_some());
418                // SAFETY: The index addition producing the end index does not overflow, it is
419                // `destination_end_index`. `offset` is also strictly smaller than count, therefore
420                // this cannot not overflow.
421                let final_destination_index = unsafe { destination_index.unchecked_add(offset) };
422
423                debug_assert!(final_source_index < len);
424                debug_assert!(final_destination_index < len);
425                // SAFETY: Both the final source and destination indices must be smaller than
426                // source_end_index and source_destination index which are smaller than len,
427                // respectively.
428                unsafe {
429                    self.copy_byte(final_source_index, final_destination_index, Ordering::Unord)
430                };
431            }
432        }
433
434        Ok(())
435    }

437    /// Copies a single byte from a given source index to some destination index.
438    ///
439    /// # Safety
440    ///
441    /// `source_index` and `destination_index` must both be less than some length previously
442    /// returned by [`Self::rd_len`].
443    unsafe fn copy_byte(&self, source_index: usize, destination_index: usize, ord: Ordering) {
444        // TODO optimize this. Currently copying one byte acquires the lock twice (rd and wr
445        // actions both acquire the lock internally).
446
447        debug_assert!(source_index.checked_add(1).is_some());
448        // SAFETY: Caller ensures that `source_index` is less than some length which must therefore
449        // be at least `source_index + 1`. Thus `source_index + 1` does not overflow.
450        let [byte] = unsafe { self.rd_data_into_array::<1>(source_index, ord, true) };
451
452        debug_assert!(destination_index.checked_add(1).is_some());
453        // SAFETY: Caller ensures that `destination_index` is less than some length which must
454        // therefore be at least `destination_index + 1`. Thus `destination_index + 1` does not
455        // overflow.
456        unsafe { self.wr_data_tearing(destination_index, iter::once(byte), ord) };
457    }

459    // Rationale behind having `source_index` and `count` when the callsite could also just create a
460    // subslice for `source_data`? Have all the index error checks in one place.
461    //
462    // <https://webassembly.github.io/spec/core/exec/instructions.html#xref-syntax-instructions-syntax-instr-memory-mathsf-memory-init-x>
463    pub fn init(
464        &self,
465        destination_index: usize,
466        source_data: &[u8],
467        source_index: usize,
468        count: usize,
469    ) -> Result<(), RuntimeError> {
470        let source_end_index = source_index
471            .checked_add(count)
472            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
473        let destination_end_index = destination_index
474            .checked_add(count)
475            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
476
477        let source_bytes = source_data
478            .get(source_index..source_end_index)
479            .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
480
481        let len = self.rd_len(Ordering::Unord);
482
483        if destination_end_index > len {
484            return Err(TrapError::MemoryOrDataAccessOutOfBounds.into());
485        }
486
487        // SAFETY: The index addition producing the end index does not overflow, it is
488        // `destination_end_index`. Also, it was checked that it is not larger than the length of
489        // the current memory.
490        unsafe {
491            self.wr_data_tearing(
492                destination_index,
493                source_bytes.iter().copied(),
494                Ordering::Unord,
495            )
496        };
497
498        Ok(())
499    }

501    /// The rd_len action.
502    ///
503    /// This action returns the length of this shared linear memory at this very specific point in
504    /// time. The returned length is not the exact length of the memory, as the memory could have
505    /// grown since reading of its length took place.
506    ///
507    /// Note that shared linear memories can only ever grow. As a result, lengths returned by this
508    /// method may only be used as a lower bound for the memory's current size for future
509    /// operations.
510    ///
511    /// # See also
512    ///
513    /// - [WebAssembly Specification 2.0 (with Threads Proposal) - 4.2.17 Events - act](https://webassembly.github.io/threads/core/exec/runtime.html#syntax-act)
514    /// - [`Self::rd_len_in_pages`]
515    pub fn rd_len(&self, ord: Ordering) -> usize {
516        // We do not use the memory ordering for the length field because it is not an atomic
517        // integer and protected by the surrounding lock.
518        let _ = ord;
519
520        let lock_guard = self.inner_data.read();
521        // Note: The spec defines a `rd` action to have a specific ord, but we can just directly
522        // read from the &usize, as no-one as write access to it anyway.
523        lock_guard.len()
524    }

532    /// The rd_data action.
533    ///
534    /// This action reads `n` bytes from `index..(index + n)` into a provided buffer where `n` is
535    /// the length of that buffer in bytes.
536    ///
537    /// # See also
538    ///
539    /// - [WebAssembly Specification 2.0 (with Threads Proposal) - 4.2.17 Events - act](https://webassembly.github.io/threads/core/exec/runtime.html#syntax-act)
540    /// - [`Self::rd_data_into_array`]
541    ///
542    /// # Safety
543    ///
544    /// - `index + n` must not overflow a usize.
545    /// - `index + n` must not be larger than some length previously returned by [`Self::rd_len`].
546    ///
547    /// `n`: length of destination buffer in bytes.
548    pub unsafe fn rd_data(
549        &self,
550        index: usize,
551        dst_byte_buffer: impl AsMut<[MaybeUninit<u8>]>,
552        ord: Ordering,
553        may_tear: bool,
554    ) {
555        if may_tear {
556            // SAFETY: The caller ensures that `index + n` does not overflow a usize and that it is
557            // not larger than a given length.
558            unsafe { self.rd_data_tearing(index, dst_byte_buffer, ord) };
559        } else {
560            // SAFETY: The caller ensures that `index + n` does not overflow a usize and that it is
561            // not larger than a given length.
562            unsafe { Self::rd_data_no_tears(self.inner_data.write(), index, dst_byte_buffer) };
563        }
564    }

566    /// A version of [`Self::rd_data`] that returns its data via an array.
567    ///
568    /// # Safety
569    ///
570    /// - `index + N` must not overflow a usize.
571    /// - `index + n` must not be larger than some length previously returned by [`Self::rd_len`].
572    pub unsafe fn rd_data_into_array<const N: usize>(
573        &self,
574        index: usize,
575        ord: Ordering,
576        may_tear: bool,
577    ) -> [u8; N] {
578        let mut bytes = [const { MaybeUninit::uninit() }; N];
579        // SAFETY: The caller ensures that `index + n` does not overflow a usize and that it is
580        // not larger than a given length.
581        unsafe { self.rd_data(index, &mut bytes, ord, may_tear) };
582
583        // Workaround for feature `maybe_uninit_uninit_array_transpose`
584        bytes.map(|byte| {
585            // SAFETY: u8 and MaybeUninit<u8> have the same layout and rd_data guarantees that all
586            // bytes passed to it are initialized properly.
587            unsafe { byte.assume_init() }
588        })
589    }
590
591    /// The tearing [`Self::rd_data`] variant.
592    ///
593    /// # Safety
594    ///
595    /// - `index + n` must not overflow a usize.
596    /// - `index + n` must not be larger than some length previously returned by [`Self::rd_len`].
597    ///
598    /// `n`: length of destination buffer in bytes.
599    unsafe fn rd_data_tearing(
600        &self,
601        index: usize,
602        mut dst_byte_buffer: impl AsMut<[MaybeUninit<u8>]>,
603        ord: Ordering,
604    ) {
605        // We can probably keep the lock guard across the entire loop. Although it might make sense
606        // to re-acquire it on every iteration for really large reads.
607        let lock_guard = self.inner_data.read();
608
609        for (offset, dst_byte) in dst_byte_buffer.as_mut().iter_mut().enumerate() {
610            debug_assert!(index.checked_add(offset).is_some());
611            // SAFETY: The caller ensures that `index + n` never overflows. `offset` is strictly
612            // smaller than `n`, and thus `index + offset` also cannot overflow.
613            let final_index = unsafe { index.unchecked_add(offset) };
614
615            debug_assert!(lock_guard.get(final_index).is_some());
616            // SAFETY: The caller ensures that `index + n <= len`. Also i is always smaller than n
617            // and thus `index + i` is always within bounds.
618            let src = unsafe { lock_guard.get_unchecked(final_index) };
619
620            let byte = src.load(ord.into_atomic_ordering());
621            dst_byte.write(byte);
622        }
623    }
624
625    /// The non-tearing [`Self::rd_data`] variant.
626    ///
627    /// This method takes a write lock guard as an argument, so that callers can potentially re-use
628    /// the lock guard.
629    ///
630    /// # Safety
631    ///
632    /// - `index + n` must not overflow a usize.
633    /// - `index + n` must not be larger than some length previously returned by [`Self::rd_len`].
634    unsafe fn rd_data_no_tears(
635        mut lock_guard: WriteLockGuard<'_, Vec<AtomicU8>>,
636        index: usize,
637        mut dst_byte_buffer: impl AsMut<[MaybeUninit<u8>]>,
638    ) {
639        for (offset, dst_byte) in dst_byte_buffer.as_mut().iter_mut().enumerate() {
640            debug_assert!(index.checked_add(offset).is_some());
641            // SAFETY: The caller ensures that `index + n` never overflows. `offset` is strictly
642            // smaller than `n`, and thus `index + offset` also cannot overflow.
643            let final_index = unsafe { index.unchecked_add(offset) };
644
645            debug_assert!(lock_guard.get(final_index).is_some());
646            // SAFETY: The caller ensures that `index + n <= len`. Also i is always smaller than n
647            // and thus `index + i` is always within bounds.
648            let src = unsafe { lock_guard.get_unchecked_mut(final_index) };
649
650            // Note: We have an exclusive reference, and therefore no need to use an atomic
651            // instruction.
652            let byte = *src.get_mut();
653            dst_byte.write(byte);
654        }
655    }

657    /// The `wr` action
658    ///
659    /// This action writes `n` bytes into the memory starting at `index`. The bytes are read from a
660    /// provided buffer in the form of an [`ExactSizeIterator`] where `n` is its length.
661    ///
662    /// # See also
663    ///
664    /// - [WebAssembly Specification 2.0 (with Threads Proposal) - 4.2.17 Events - act](https://webassembly.github.io/threads/core/exec/runtime.html#syntax-act)
665    /// - [`Self::rd_data_into_array`]
666    ///
667    /// # Safety
668    ///
669    /// - `index + n` must not overflow a usize.
670    /// - `index + n` must not be larger than some length previously returned by [`Self::rd_len`].
671    ///
672    /// `n`: length of source buffer in bytes.
673    pub unsafe fn wr_data(
674        &self,
675        index: usize,
676        bytes: impl ExactSizeIterator<Item = u8>,
677        ord: Ordering,
678        may_tear: bool,
679    ) {
680        debug_assert!(index.checked_add(bytes.len()).is_some());
681
682        if may_tear {
683            // SAFETY: The caller ensures that `index + n` does not overflow a usize and that it is
684            // not larger than a given length.
685            unsafe { self.wr_data_tearing(index, bytes, ord) }
686        } else {
687            // SAFETY: The caller ensures that `index + n` does not overflow a usize and that it is
688            // not larger than a given length.
689            unsafe { Self::wr_data_no_tears(self.inner_data.write(), index, bytes) }
690        }
691    }
692
693    /// The tearing [`Self::wr_data`] variant.
694    ///
695    /// # Safety
696    ///
697    /// - `index + n` must not overflow a usize.
698    /// - `index + n` must be less or equal to some length previously returned by [`Self::rd_len`].
699    ///
700    /// `n`: length of iterator over source buffer bytes.
701    unsafe fn wr_data_tearing(
702        &self,
703        index: usize,
704        bytes: impl ExactSizeIterator<Item = u8>,
705        ord: Ordering,
706    ) {
707        let lock_guard = self.inner_data.read();
708
709        for (offset, byte) in bytes.enumerate() {
710            debug_assert!(index.checked_add(offset).is_some());
711            // SAFETY: The caller ensures that `index + n` never overflows. `offset` is strictly
712            // smaller than `n`, and thus `index + offset` also cannot overflow.
713            let final_index = unsafe { index.unchecked_add(offset) };
714
715            debug_assert!(lock_guard.get(final_index).is_some());
716            // SAFETY: The caller ensures that `index + n <= len`. Also i is always smaller than n
717            // and thus `index + i` is always within bounds.
718            let dst = unsafe { lock_guard.get_unchecked(final_index) };
719
720            dst.store(byte, ord.into_atomic_ordering());
721        }
722    }
723
724    /// The non-tearing [`Self::wr_data`] variant.
725    ///
726    /// # Safety
727    ///
728    /// - `index + n` must not overflow a usize.
729    /// - `index + n` must not be larger than some length previously returned by [`Self::rd_len`].
730    ///
731    /// `n`: length of iterator over source buffer bytes.
732    unsafe fn wr_data_no_tears(
733        mut lock_guard: WriteLockGuard<'_, Vec<AtomicU8>>,
734        index: usize,
735        bytes: impl ExactSizeIterator<Item = u8>,
736    ) {
737        for (offset, byte) in bytes.enumerate() {
738            debug_assert!(index.checked_add(offset).is_some());
739            // SAFETY: The caller ensures that `index + n` never overflows. `offset` is strictly
740            // smaller than `n`, and thus `index + offset` also cannot overflow.
741            let final_index = unsafe { index.unchecked_add(offset) };
742
743            debug_assert!(lock_guard.get(final_index).is_some());
744            // SAFETY: The caller ensures that `index + n <= len`. Also i is always smaller than n
745            // and thus `index + i` is always within bounds.
746            let dst = unsafe { lock_guard.get_unchecked_mut(final_index) };
747
748            // Note: We have an exclusive reference, and therefore no need to use an atomic
749            // instruction.
750            *dst.get_mut() = byte;
751        }
752    }
753    /// Allows a given closure to temporarily access the entire memory as a `&mut [u8]`.
754    ///
755    /// # Note on locking
756    ///
757    /// This operation exclusively locks the entire linear memory temporarily. This operation blocks
758    /// until the lock is successfully acquired.
759    pub fn access_mut_slice<R>(&self, accessor: impl FnOnce(&mut [u8]) -> R) -> R {
760        let mut write_lock_guard = self.inner_data.write();
761        let non_atomic_slice = atomic_u8_get_mut_slice(&mut write_lock_guard);
762        accessor(non_atomic_slice)
763    }
764}

766impl core::fmt::Debug for SharedLinearMemory {
767    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
768        /// A helper struct for formatting a [`Vec<UnsafeCell<u8>>`] which is guarded by a [`ReadLockGuard`].
769        /// This formatter is able to detect and format byte repetitions in a compact way.
770        struct RepetitionDetectingMemoryWriter<'a>(ReadLockGuard<'a, Vec<AtomicU8>>);
771        impl core::fmt::Debug for RepetitionDetectingMemoryWriter<'_> {
772            fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
773                /// The number of repetitions required for successive elements to be grouped
774                // together.
775                const MIN_REPETITIONS_FOR_GROUP: usize = 8;
776
777                // First we create an iterator over all bytes
778                let mut bytes = self
779                    .0
780                    .iter()
781                    .map(|x| x.load(sync::atomic::Ordering::Relaxed));
782
783                // Then we iterate over all bytes and deduplicate repetitions. This produces an
784                // iterator of pairs, consisting of the number of repetitions and the repeated byte
785                // itself. `current_group` is captured by the iterator and used as state to track
786                // the current group.
787                let mut current_group: Option<(usize, u8)> = None;
788                let deduplicated_with_count = iter::from_fn(|| {
789                    for byte in bytes.by_ref() {
790                        // If the next byte is different than the one being tracked currently...
791                        if current_group.is_some() && current_group.unwrap().1 != byte {
792                            // ...then end and emit the current group but also start a new group for
793                            // the next byte with an initial count of 1.
794                            return current_group.replace((1, byte));
795                        }
796                        // Otherwise increment the current group's counter or start a new group if
797                        // this was the first byte.
798                        current_group.get_or_insert((0, byte)).0 += 1;
799                    }
800                    // In the end when there are no more bytes to read, directly emit the last
801                    current_group.take()
802                });
803
804                // Finally we use `DebugList` to print a list of all groups, while writing out all
805                // elements from groups with less than `MIN_REPETITIONS_FOR_GROUP` elements.
806                let mut list = f.debug_list();
807                deduplicated_with_count.for_each(|(count, value)| {
808                    if count < MIN_REPETITIONS_FOR_GROUP {
809                        list.entries(iter::repeat_n(value, count));
810                    } else {
811                        list.entry(&format_args!("#{count} × {value}"));
812                    }
813                });
814                list.finish()
815            }
816        }
817
818        // Format the linear memory by using Rust's formatter helpers and the previously defined
819        // `RepetitionDetectingMemoryWriter`
820        f.debug_struct("SharedLinearMemory")
821            .field(
822                "inner_data",
823                &RepetitionDetectingMemoryWriter(self.inner_data.read()),
824            )
825            .finish()
826    }
827}

829#[inline(always)]
830fn tearing<const N: usize, T: LittleEndianBytes<N>>(index: usize) -> bool {
831    let is_properly_aligned = index.is_multiple_of(N);
832    let is_max_32_bits_large = N <= 4;
833    let no_tears = is_properly_aligned && is_max_32_bits_large;
834    !no_tears
835}

837/// Converts an exclusively borrowed slice of atomic `u8`s to a slice of
838/// non-atomic `u8`s
839// TODO when `atomic_from_mut` is stabilized, replace this function with
840// `Atomic::U8::get_mut_slice`
841fn atomic_u8_get_mut_slice(slice: &mut [AtomicU8]) -> &mut [u8] {
842    // SAFETY: the mutable reference guarantees unique ownership
843    unsafe { &mut *(slice as *mut [AtomicU8] as *mut [u8]) }
844}
