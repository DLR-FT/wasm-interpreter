17            .finish_non_exhaustive()

21impl MemInst {
22    /// If the grow is successful, the previous length of the memory is returned in pages.
23    ///
24    /// See: [WebAssembly Specification 2.0 - 4.5.3.9 Growing Memories](https://www.w3.org/TR/2025/CRD-wasm-core-2-20250616/#growing-memories%E2%91%A0)
25    pub fn grow<T: Config>(&mut self, n: u32) -> Result<u32, RuntimeError> {
26        // 1. Let meminst be the memory instance to grow and n the number of pages by which to grow
27        //    it.
28        //
29        // `self` is the meminst and `n` the number of pages by which to grow it.
30
31        // 2. Assert: The length of meminst.data is divisible by the page size 64Ki.
32        // 3. Let len be n added to the length of meminst.data divided by the page size 64Ki.
33        let previous_len = self.len_pages();
34        let len = u32::from(previous_len).checked_add(n);
35
36        // 4. If len is larger than 2^16, then fail.
37        //
38        // Checks if it would have been larger than 2^64
39        let Some(len) = len else {
40            return Err(RuntimeError::MemoryOverflowed);
41        };
42        // Check if it is larger than 2^16
43        let len = u16::try_from(len).map_err(|_| RuntimeError::MemoryOverflowed)?;
44
45        // 5. Let limits be the structure of memory type meminst.type.
46        let limits = self.ty.limits;
47
48        // 6. Let limits' be limits with min updated to len.
49        let limits_prime = Limits {
50            min: u32::from(len),
51            ..limits
52        };
53
54        // 7. If limits' is not valid, then fail.
55        if limits_prime.max.is_some_and(|max| limits_prime.min > max) {
56            return Err(RuntimeError::MemoryGrowExceededLimit);
57        }
58
59        // 8. Append n times 64Ki bytes with value 0x00 to meminst.data.
60        //
61        // For us this operation is fallible, as a custom upper size limit can be set through
62        // `Config`.
63        self.mem
64            .grow::<T>(n.into_usize())
65            .map_err(|MemorySizeOverflow| RuntimeError::MemoryOverflowed)?;
66
67        // 9. Set meminst.type to the memory type limits'.
68        self.ty.limits = limits_prime;
69
70        // Additionally, return the previous length
71        Ok(u32::from(previous_len))
72    }
73
74    pub fn len_pages(&self) -> u16 {
75        self.mem
76            .len_pages()
77            .try_into()
78            .expect("memory length is pages is never 2^16 or larger")
79    }
80}
