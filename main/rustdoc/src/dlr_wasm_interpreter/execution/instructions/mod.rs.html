52/// A non-error outcome of interpretation
53pub enum InterpreterLoopOutcome {
54    /// Execution has returned normally, i.e. the end of the bottom-most function on the stack was
55    /// reached. The return values for the initially invoked function are on the stack.
56    ExecutionReturned,
57    /// Execution was preempted because there was not enough fuel in the [`WasmResumable`] object.
58    OutOfFuel {
59        /// The amount of fuel required to continue execution at least the next instruction.
60        required_fuel: NonZeroU64,
61    },
62    /// A host function instance was called. The arguments for the host function call have been
63    /// collected into `params` already.
64    HostCalled {
65        func_addr: FuncAddr,
66        // TODO this allocation might be preventable. mutably borrow the stack instead
67        params: Vec<Value>,
68        hostcode: Hostcode,
69    },
70}

72/// The execution state interacted with by all instructions.
73///
74/// # Safety
75///
76/// - The [`WasmDecoder`] must point to the Wasm code for the module of the current module instance.
77/// - The [`WasmDecoder`] must point into Wasm code of the current function as set in

79/// - The [`StoreInner`] must be valid.
80/// - The [`WasmResumable`] must be valid in [`StoreInner`].
81/// - All address types contained in this struct must be valid in the [`StoreInner`].
82/// - The current sidetable must be correct for the module of the current module instance.
83/// - The end marker for the current function must point to the end index of the current function in
84///   the current module's bytecode.

87    wasm: &'a mut WasmDecoder<'wasm>,
88    resumable: &'a mut WasmResumable,

94}

96//helper function for avoiding code duplication at intraprocedural jumps
97fn do_sidetable_control_transfer(
98    wasm: &mut WasmDecoder,
99    stack: &mut Stack,
100    current_stp: &mut usize,
101    current_sidetable: &Sidetable,
102) -> Result<(), RuntimeError> {
103    let sidetable_entry = &current_sidetable[*current_stp];
104
105    stack.remove_in_between(sidetable_entry.popcnt, sidetable_entry.valcnt);
106
107    *current_stp = sidetable_entry.stp;
108    wasm.pc = sidetable_entry.pc;
109
110    Ok(())
111}

113#[inline(always)]
114fn calculate_mem_address(memarg: &MemArg, relative_address: u32) -> Result<usize, RuntimeError> {
115    // The spec states that this should be a 33 bit integer, e.g. it is not legal to wrap if the
116    // sum of offset and relative_address exceeds u32::MAX. To emulate this behavior, we use a
117    // checked addition.
118    // See: https://webassembly.github.io/spec/core/syntax/instructions.html#memory-instructions
119    let effective_address = memarg
120        .offset
121        .checked_add(relative_address)
122        .ok_or(TrapError::MemoryOrDataAccessOutOfBounds)?;
123
124    Ok(effective_address.into_usize())
125}

128/// # Safety
129///
130/// 1. The module address `current_module` must be valid in `store_modules` for a module instance `module_inst`.
131/// 2. The table index `table_idx` must be valid in `module_inst` for a table address `table_addr`.
132/// 3. `table_addr` must be valid in `store_tables`.
133/// 4. The element index `elem_idx` must be valid in `module_inst` for an element address `elem_addr`.
134/// 5. `elem_addr` must be valid in `store_elements`.
135// TODO instead of passing all module instances and the current module addr
136// separately, directly pass a `&ModuleInst`.
137#[inline(always)]
138#[allow(clippy::too_many_arguments)]
139pub(super) unsafe fn table_init(
140    store_modules: &AddrVec<ModuleAddr, ModuleInst>,
141    store_tables: &mut AddrVec<TableAddr, TableInst>,
142    store_elements: &AddrVec<ElemAddr, ElemInst>,
143    current_module: ModuleAddr,
144    elem_idx: ElemIdx,
145    table_idx: TableIdx,
146    n: u32,
147    s: i32,
148    d: i32,
149) -> Result<(), RuntimeError> {
150    let n = n.into_usize();
151    let s = s.cast_unsigned().into_usize();
152    let d = d.cast_unsigned().into_usize();
153
154    // SAFETY: The caller ensures that this module address is valid in this
155    // address vector (1).
156    let module_inst = unsafe { store_modules.get(current_module) };
157    // SAFETY: The caller ensures that `table_idx` is valid for this specific
158    // `IdxVec` (2).
159    let table_addr = *unsafe { module_inst.table_addrs.get(table_idx) };
160    // SAFETY: The caller ensures that `elem_idx` is valid for this specific
161    // `IdxVec` (4).
162    let elem_addr = *unsafe { module_inst.elem_addrs.get(elem_idx) };
163    // SAFETY: The caller ensures that this table address is valid in this
164    // address vector (3).
165    let tab = unsafe { store_tables.get_mut(table_addr) };
166    // SAFETY: The caller ensures that this element address is valid in this
167    // address vector (5).
168    let elem = unsafe { store_elements.get(elem_addr) };
169
170    let final_src_offset = s
171        .checked_add(n)
172        .filter(|&res| res <= elem.len())
173        .ok_or(TrapError::TableOrElementAccessOutOfBounds)?;
174
175    if d.checked_add(n)
176        .filter(|&res| res <= tab.len().into_usize())
177        .is_none()
178    {
179        return Err(TrapError::TableOrElementAccessOutOfBounds.into());
180    }
181
182    let dest = &mut tab.elem[d..];
183    let src = &elem.references[s..final_src_offset];
184    dest[..src.len()].copy_from_slice(src);
185    Ok(())
186}

216/// # Safety
217///
218/// 1. The module address `current_module` must be valid in `store_modules` for some module instance `module_inst`.
219/// 2. The memory index `mem_idx` must be valid in `module_inst` for some memory address `mem_addr`.
220/// 3. `mem_addr` must be valid in `store_memories` for some memory instance `mem`.
221/// 4. The data index `data_idx` must be valid in `module_inst` for some data address `data_addr`.
222/// 5. `data_addr` must be valid in `store_data`.
223#[inline(always)]
224#[allow(clippy::too_many_arguments)]
225pub(super) unsafe fn memory_init(
226    store_modules: &AddrVec<ModuleAddr, ModuleInst>,
227    store_memories: &mut AddrVec<MemAddr, MemInst>,
228    store_data: &AddrVec<DataAddr, DataInst>,
229    current_module: ModuleAddr,
230    data_idx: DataIdx,
231    mem_idx: MemIdx,
232    n: u32,
233    s: u32,
234    d: u32,
235) -> Result<(), RuntimeError> {
236    let n = n.into_usize();
237    let s = s.into_usize();
238    let d = d.into_usize();
239
240    // SAFETY: The caller ensures that this is module address is valid in this
241    // address vector (1).
242    let module_inst = unsafe { store_modules.get(current_module) };
243    // SAFETY: The caller ensures that `mem_idx` is valid for this specific
244    // `IdxVec` (2).
245    let mem_addr = *unsafe { module_inst.mem_addrs.get(mem_idx) };
246    // SAFETY: The caller ensures that this memory address is valid in this
247    // address vector (3).
248    let mem = unsafe { store_memories.get_mut(mem_addr) };
249    // SAFETY: The caller ensures that `data_idx` is valid for this specific
250    // `IdxVec` (4).
251    let data_addr = *unsafe { module_inst.data_addrs.get(data_idx) };
252    // SAFETY: The caller ensures that this data address is valid in this
253    // address vector (5).
254    let data = unsafe { store_data.get(data_addr) };
255
256    match mem {
257        MemInst::Unshared(unshared_mem) => {
258            unshared_mem.mem.init(d, &data.data, s, n)?;
259        }
260        MemInst::Shared(shared_mem) => {
261            shared_mem.mem.init(d, &data.data, s, n)?;
262        }
263    }
264
265    Ok(())
266}

300#[inline(always)]
301pub(crate) fn to_lanes<const M: usize, const N: usize, T: LittleEndianBytes<M>>(
302    data: [u8; 16],
303) -> [T; N] {
304    assert_eq!(M * N, 16);
305
306    let mut lanes = data
307        .chunks(M)
308        .map(|chunk| T::from_le_bytes(chunk.try_into().unwrap()));
309    array::from_fn(|_| lanes.next().unwrap())
310}

312#[inline(always)]
313pub(crate) fn from_lanes<const M: usize, const N: usize, T: LittleEndianBytes<M>>(
314    lanes: [T; N],
315) -> [u8; 16] {
316    assert_eq!(M * N, 16);
317
318    let mut bytes = lanes.into_iter().flat_map(T::to_le_bytes);
319    array::from_fn(|_| bytes.next().unwrap())
320}

322#[inline(always)]
323fn decrement_fuel(cost: u64, maybe_fuel: &mut Option<u64>) -> ControlFlow<InterpreterLoopOutcome> {
324    if let Some(fuel) = maybe_fuel {
325        if *fuel >= cost {
326            *fuel -= cost;
327        } else {
328            return ControlFlow::Break(InterpreterLoopOutcome::OutOfFuel {
329                required_fuel: NonZeroU64::new(cost - *fuel)
330                    .expect("the last check guarantees that the current fuel is smaller than cost"),
331            });
332        }
333    }
334
335    ControlFlow::Continue(())
336}
