17#[inline(always)]
18pub unsafe fn ref_null(state: State) -> Result<ControlFlow<InterpreterLoopOutcome>, RuntimeError> {
19    let reftype = RefType::decode(state.wasm).unwrap_validated();
20
21    state
22        .resumable
23        .stack
24        .push_value(Value::Ref(Ref::Null(reftype)))?;
25    Ok(ControlFlow::Continue(()))
26}

28#[inline(always)]
29pub unsafe fn ref_is_null(
30    state: State,
31) -> Result<ControlFlow<InterpreterLoopOutcome>, RuntimeError> {
32    // SAFETY: Validation guarantees that there is a ref value on the stack.
33    let rref: Ref = unsafe { state.resumable.stack.pop_value() }
34        .try_into()
35        .unwrap_validated();
36    let is_null = matches!(rref, Ref::Null(_));
37
38    let res = if is_null { 1 } else { 0 };
39    state.resumable.stack.push_value(Value::I32(res))?;
40    Ok(ControlFlow::Continue(()))
41}

43// https://webassembly.github.io/spec/core/exec/instructions.html#xref-syntax-instructions-syntax-instr-ref-mathsf-ref-func-x
44#[inline(always)]
45pub unsafe fn ref_func(state: State) -> Result<ControlFlow<InterpreterLoopOutcome>, RuntimeError> {
46    // SAFETY: Validation guarantees a valid function index to be
47    // next.
48    let func_idx = unsafe { FuncIdx::decode_unchecked(state.wasm) };
49
50    // SAFETY: The current module address must come from the current
51    // store, because it is the only parameter to this function that
52    // can contain module addresses. All stores guarantee all
53    // addresses in them to be valid within themselves.
54    let current_module = unsafe { state.modules.get(*state.current_module) };
55    // SAFETY: Validation guarantees the function index to be valid
56    // in the current module.
57    let func_addr = unsafe { current_module.func_addrs.get(func_idx) };
58    state
59        .resumable
60        .stack
61        .push_value(Value::Ref(Ref::Func(*func_addr)))?;
62    Ok(ControlFlow::Continue(()))
63}
