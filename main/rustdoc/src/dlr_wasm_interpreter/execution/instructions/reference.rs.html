17define_instruction!(super::ref_null, ref_null_mod, fuel_check = flat(REF_NULL));
18#[inline(always)]
19pub unsafe fn ref_null(state: State) -> Result<ControlFlow<InterpreterLoopOutcome>, RuntimeError> {
20    let reftype = RefType::decode(state.wasm).unwrap_validated();
21
22    state
23        .resumable
24        .stack
25        .push_value(Value::Ref(Ref::Null(reftype)))?;
26    trace!("Instruction: ref.null '{:?}' -> [{:?}]", reftype, reftype);
27    Ok(ControlFlow::Continue(()))
28}

30define_instruction!(
31    super::ref_is_null,
32    ref_is_null_mod,
33    fuel_check = flat(REF_IS_NULL)
34);
35#[inline(always)]
36pub unsafe fn ref_is_null(
37    state: State,
38) -> Result<ControlFlow<InterpreterLoopOutcome>, RuntimeError> {
39    let rref: Ref = state
40        .resumable
41        .stack
42        .pop_value()
43        .try_into()
44        .unwrap_validated();
45    let is_null = matches!(rref, Ref::Null(_));
46
47    let res = if is_null { 1 } else { 0 };
48    trace!("Instruction: ref.is_null [{}] -> [{}]", rref, res);
49    state.resumable.stack.push_value(Value::I32(res))?;
50    Ok(ControlFlow::Continue(()))
51}

53// https://webassembly.github.io/spec/core/exec/instructions.html#xref-syntax-instructions-syntax-instr-ref-mathsf-ref-func-x
54define_instruction!(super::ref_func, ref_func_mod, fuel_check = flat(REF_FUNC));
55#[inline(always)]
56pub unsafe fn ref_func(state: State) -> Result<ControlFlow<InterpreterLoopOutcome>, RuntimeError> {
57    // SAFETY: Validation guarantees a valid function index to be
58    // next.
59    let func_idx = unsafe { FuncIdx::decode_unchecked(state.wasm) };
60
61    // SAFETY: The current module address must come from the current
62    // store, because it is the only parameter to this function that
63    // can contain module addresses. All stores guarantee all
64    // addresses in them to be valid within themselves.
65    let current_module = unsafe { state.modules.get(*state.current_module) };
66    // SAFETY: Validation guarantees the function index to be valid
67    // in the current module.
68    let func_addr = unsafe { current_module.func_addrs.get(func_idx) };
69    state
70        .resumable
71        .stack
72        .push_value(Value::Ref(Ref::Func(*func_addr)))?;
73    Ok(ControlFlow::Continue(()))
74}
