5/// Trait that allows user specified configuration for various items during interpretation. Additionally, the types
6/// implementing this trait can act as custom user data within an interpreter instance, passed along to each method of
7/// this trait and host functions whenever they are invoked.
8///
9/// The default implementation of all trait methods have the least overhead, i. e. most can be optimized out fully.
10// It must always be checked that there is no additional performance penalty for the default config!
11pub trait Config {
12    /// Maximum number of values in the value stack
13    ///
14    /// The default of `0xd0`/`208` is just the bare minimum multiple of 16 to pass the Specification Test Suite
15    const MAX_VALUE_STACK_SIZE: usize = 0xd0;
16
17    /// Maximum number of cascading function invocations
18    ///
19    /// The default of `0xd0`/`208` is just the bare minimum multiple of 16 to pass the Specification Test Suite
20    const MAX_CALL_STACK_SIZE: usize = 0xd0;
21
22    /// An optional limit for the number of pages a memory's size can grow to.
23    // TODO(memory64): Use Option<NonZeroUsize> with limit of 2^48 pages
24    // TODO(custom-page-sizes): Use Option<NonZeroU32>
25    const MAX_NUMBER_OF_MEMORY_PAGES: Option<NonZeroU16> = None;
26
27    /// An optional limit for the number of elements a table's size can grow to.
28    const MAX_NUMBER_OF_TABLE_ELEMENTS: Option<NonZeroUsize> = None;
29
30    /// The mechanism to use for dispatching during interpretation. Refer to [`DispatchMechanism`]
31    /// for a list of all mechanisms, including their up- and downsides.
32    const DISPATCH_MECHANISM: DispatchMechanism = DispatchMechanism::LoopCall;
33
34    /// A hook which is called before every wasm instruction
35    ///
36    /// This allows the most intricate insight into the interpreters behavior, at the cost of a
37    /// hefty performance penalty
38    #[inline(always)]
39    fn instruction_hook(&mut self, _bytecode: &[u8], _pc: usize) {}
40
41    /// Amount of fuel to be deducted when a single byte `instr` is hit. The cost corresponding to `UNREACHABLE` and
42    /// `END` instructions and other bytes that do not correspond to any Wasm instruction are ignored.
43    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
44    // yields constants.
45    #[inline(always)]
46    fn get_flat_cost(_instr: u8) -> u64 {
47        1
48    }
49
50    /// Amount of fuel to be deducted when a multi-byte instruction that starts with the byte 0xFC is hit. This method
51    /// should return the cost of an instruction obtained by prepending 0xFC to of an unsigned 32-bit LEB
52    /// representation of `instr`. Multi-byte sequences obtained this way that do not correspond to any Wasm instruction
53    /// are ignored.
54    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
55    // yields constants.
56    #[inline(always)]
57    fn get_fc_extension_flat_cost(_instr: u32) -> u64 {
58        1
59    }
60
61    /// Amount of fuel to be deducted when a multi-byte instruction that starts with the byte 0xFD is hit. This method
62    /// should return the cost of an instruction obtained by prepending 0xFD to of an unsigned 32-bit LEB
63    /// representation of `instr`. Multi-byte sequences obtained this way that do not correspond to any Wasm instruction
64    /// are ignored.
65    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
66    // yields constants.
67    #[inline(always)]
68    fn get_fd_extension_flat_cost(_instr: u32) -> u64 {
69        1
70    }
71
72    /// Amount of fuel to be deducted per element of a single byte instruction `instr` that executes in asymptotically
73    /// linear time with respect to one of the values it pops from the stack.
74    ///
75    /// In Wasm 2.0 specification, this applies to the following instructions:
76    /// - `MEMORY.GROW` of type `[n: i32] -> [i32]`
77    ///
78    /// The cost of the instruction is calculated as `cost := get_flat_cost(instr) + n*get_cost_per_element(instr)`.
79    /// where `n` is the stack value marked in the instruction type signature above. Other instructions and bytes that
80    /// do not correspond to any instruction are ignored.
81    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
82    // yields constants.
83    #[inline(always)]
84    fn get_cost_per_element(_instr: u8) -> u64 {
85        0
86    }
87
88    /// Amount of fuel to be deducted per element of a  multi-byte instruction that starts with the byte 0xFC,
89    /// which executes in asymptotically linear time with respect to one of the values it pops from the stack. This
90    /// method should return the cost of an instruction obtained by prepending 0xFD to of an unsigned 32-bit LEB
91    /// representation of `instr`. Multi-byte sequences obtained this way that do not correspond to any Wasm instruction
92    /// are ignored.
93    ///
94    /// In Wasm 2.0 specification, this applies to the following instructions:
95    /// - `MEMORY.INIT x`  of type `[d:i32 s: i32 n: i32] -> []`
96    /// - `MEMORY.FILL`    of type `[d: i32 val: i32 n: i32] -> []`
97    /// - `MEMORY.COPY`    of type `[d: i32 s: i32 n: i32] -> []`
98    /// - `TABLE.GROW x`   of type `[val: ref n: i32] -> [i32]`
99    /// - `TABLE.INIT x y` of type `[d: i32 s: i32 n: i32] -> []`
100    /// - `TABLE.FILL x`   of type `[i: i32 val: ref n: i32] -> []`
101    /// - `TABLE.COPY x y` of type `[d: i32 s: i32 n: i32] -> []`
102    ///
103    /// The cost of the instruction is calculated as `cost := get_flat_cost(instr) + n*get_cost_per_element(instr)`.
104    /// where `n` is the stack value marked in the instruction type signature above. Other instructions and multi-byte
105    /// sequences that do not correspond to any instruction are ignored.
106    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
107    // yields constants.
108    #[inline(always)]
109    fn get_fc_extension_cost_per_element(_instr: u32) -> u64 {
110        0
111    }
112}

114/// Default implementation of the interpreter configuration, with all hooks empty
115impl Config for () {}

117/// Return the number of bytes consumed for stack memory.
118///
119/// This includes all bytes allocated for Call-Stack and Value-Stack, including management
120/// structures (i.e. the fat pointers holding the allocations of the aforementioned Call- and
121/// Value-Stack).
122pub const fn stack_memory_bytes_total<T: Config>() -> usize {
123    use crate::{
124        execution::runtime_structure::value_stack::{CallFrame, Stack},
125        Value,
126    };
127
128    let value_stack_size = <T as Config>::MAX_VALUE_STACK_SIZE * core::mem::size_of::<Value>();
129    let call_stack_size = <T as Config>::MAX_CALL_STACK_SIZE * core::mem::size_of::<CallFrame>();
130    let stack_struct_size = core::mem::size_of::<Stack>();
131
132    value_stack_size + call_stack_size + stack_struct_size
133}

135#[cfg(test)]
136mod test {
137    use crate::{
138        execution::runtime_structure::value_stack::{CallFrame, Stack},
139        Value,
140    };
141
142    use super::*;
143
144    #[test]
145    fn size_of_stack_simple() {
146        let management_only_stack_size = core::mem::size_of::<Stack>();
147        let default_total_stack_size = stack_memory_bytes_total::<()>();
148        let default_data_only_value_stack_size =
149            <() as Config>::MAX_VALUE_STACK_SIZE * core::mem::size_of::<Value>();
150        let default_data_only_call_stack_size =
151            <() as Config>::MAX_CALL_STACK_SIZE * core::mem::size_of::<CallFrame>();
152
153        let default_data_only_stack_size =
154            default_data_only_call_stack_size + default_data_only_value_stack_size;
155        assert_eq!(
156            default_data_only_stack_size as usize,
157            default_total_stack_size - management_only_stack_size
158        );
159    }
