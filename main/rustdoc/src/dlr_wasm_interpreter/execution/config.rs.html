5/// Trait that allows user specified configuration for various items during interpretation. Additionally, the types
6/// implementing this trait can act as custom user data within an interpreter instance, passed along to each method of
7/// this trait and host functions whenever they are invoked.
8///
9/// The default implementation of all trait methods have the least overhead, i. e. most can be optimized out fully.
10// It must always be checked that there is no additional performance penalty for the default config!
11pub trait Config {
12    /// Maximum number of values in the value stack
13    const MAX_VALUE_STACK_SIZE: usize = 0x10000; // 64 Kibi-Values
14
15    /// Maximum number of cascading function invocations
16    const MAX_CALL_STACK_SIZE: usize = 0x1000; // 4 Kibi-Functions
17
18    /// An optional limit for the number of pages a memory's size can grow to.
19    // TODO(memory64): Use Option<NonZeroUsize> with limit of 2^48 pages
20    // TODO(custom-page-sizes): Use Option<NonZeroU32>
21    const MAX_NUMBER_OF_MEMORY_PAGES: Option<NonZeroU16> = None;
22
23    /// An optional limit for the number of elements a table's size can grow to.
24    const MAX_NUMBER_OF_TABLE_ELEMENTS: Option<NonZeroUsize> = None;
25
26    /// The mechanism to use for dispatching during interpretation. Refer to [`DispatchMechanism`]
27    /// for a list of all mechanisms, including their up- and downsides.
28    const DISPATCH_MECHANISM: DispatchMechanism = DispatchMechanism::LoopCall;
29
30    /// A hook which is called before every wasm instruction
31    ///
32    /// This allows the most intricate insight into the interpreters behavior, at the cost of a
33    /// hefty performance penalty
34    #[inline(always)]
35    fn instruction_hook(&mut self, _bytecode: &[u8], _pc: usize) {}
36
37    /// Amount of fuel to be deducted when a single byte `instr` is hit. The cost corresponding to `UNREACHABLE` and
38    /// `END` instructions and other bytes that do not correspond to any Wasm instruction are ignored.
39    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
40    // yields constants.
41    #[inline(always)]
42    fn get_flat_cost(_instr: u8) -> u64 {
43        1
44    }
45
46    /// Amount of fuel to be deducted when a multi-byte instruction that starts with the byte 0xFC is hit. This method
47    /// should return the cost of an instruction obtained by prepending 0xFC to of an unsigned 32-bit LEB
48    /// representation of `instr`. Multi-byte sequences obtained this way that do not correspond to any Wasm instruction
49    /// are ignored.
50    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
51    // yields constants.
52    #[inline(always)]
53    fn get_fc_extension_flat_cost(_instr: u32) -> u64 {
54        1
55    }
56
57    /// Amount of fuel to be deducted when a multi-byte instruction that starts with the byte 0xFD is hit. This method
58    /// should return the cost of an instruction obtained by prepending 0xFD to of an unsigned 32-bit LEB
59    /// representation of `instr`. Multi-byte sequences obtained this way that do not correspond to any Wasm instruction
60    /// are ignored.
61    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
62    // yields constants.
63    #[inline(always)]
64    fn get_fd_extension_flat_cost(_instr: u32) -> u64 {
65        1
66    }
67
68    /// Amount of fuel to be deducted per element of a single byte instruction `instr` that executes in asymptotically
69    /// linear time with respect to one of the values it pops from the stack.
70    ///
71    /// In Wasm 2.0 specification, this applies to the following instructions:
72    /// - `MEMORY.GROW` of type `[n: i32] -> [i32]`
73    ///
74    /// The cost of the instruction is calculated as `cost := get_flat_cost(instr) + n*get_cost_per_element(instr)`.
75    /// where `n` is the stack value marked in the instruction type signature above. Other instructions and bytes that
76    /// do not correspond to any instruction are ignored.
77    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
78    // yields constants.
79    #[inline(always)]
80    fn get_cost_per_element(_instr: u8) -> u64 {
81        0
82    }
83
84    /// Amount of fuel to be deducted per element of a  multi-byte instruction that starts with the byte 0xFC,
85    /// which executes in asymptotically linear time with respect to one of the values it pops from the stack. This
86    /// method should return the cost of an instruction obtained by prepending 0xFD to of an unsigned 32-bit LEB
87    /// representation of `instr`. Multi-byte sequences obtained this way that do not correspond to any Wasm instruction
88    /// are ignored.
89    ///
90    /// In Wasm 2.0 specification, this applies to the following instructions:
91    /// - `MEMORY.INIT x`  of type `[d:i32 s: i32 n: i32] -> []`
92    /// - `MEMORY.FILL`    of type `[d: i32 val: i32 n: i32] -> []`
93    /// - `MEMORY.COPY`    of type `[d: i32 s: i32 n: i32] -> []`
94    /// - `TABLE.GROW x`   of type `[val: ref n: i32] -> [i32]`
95    /// - `TABLE.INIT x y` of type `[d: i32 s: i32 n: i32] -> []`
96    /// - `TABLE.FILL x`   of type `[i: i32 val: ref n: i32] -> []`
97    /// - `TABLE.COPY x y` of type `[d: i32 s: i32 n: i32] -> []`
98    ///
99    /// The cost of the instruction is calculated as `cost := get_flat_cost(instr) + n*get_cost_per_element(instr)`.
100    /// where `n` is the stack value marked in the instruction type signature above. Other instructions and multi-byte
101    /// sequences that do not correspond to any instruction are ignored.
102    // It must always be checked that the calls to this method fold into a constant if it is just a match statement that
103    // yields constants.
104    #[inline(always)]
105    fn get_fc_extension_cost_per_element(_instr: u32) -> u64 {
106        0
107    }
108}

110/// Default implementation of the interpreter configuration, with all hooks empty
111impl Config for () {}
