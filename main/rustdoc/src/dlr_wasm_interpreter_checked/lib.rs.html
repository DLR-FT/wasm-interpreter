1//! A safe API wrapper for [`Store`](dlr_wasm_interpreter::Store) and other utility crates that can
2//! be enabled through the features:
3//!
4//! - `linker`
5//! - `interop`
6//!
7//! All extension methods defined in this module use special _stored_ objects.  These objects are
8//! essentially normal objects like [`FuncAddr`](dlr_wasm_interpreter::FuncAddr),
9//! [`RunState`](dlr_wasm_interpreter::RunState) or [`Value`](dlr_wasm_interpreter::Value). However,
10//! they also contain an additional field of type [`StoreId`] as a tag to know to which
11//! [`Store`](dlr_wasm_interpreter::Store) they belong to.
12//!
13//! While this is easy for address types like [`FuncAddr`](dlr_wasm_interpreter::FuncAddr) or
14//! [`MemAddr`](dlr_wasm_interpreter::MemAddr), some types are enums and their variants are visible
15//! to the user. For example, consider the [`Value`](dlr_wasm_interpreter::Value) enum, where users
16//! have full access to all of its variants. To be able to attach a tag only to the
17//! [`Value::Ref`](dlr_wasm_interpreter::Value::Ref) variant of this enum, the entire enum has to be
18//! re-defined. The result is a completely new type [`StoredValue`].
19

46/// A unique identifier for a specific [`Store`]
47#[derive(Copy, Clone, Debug, PartialEq, Eq)]
48pub struct StoreId(u64);
49
50impl StoreId {
51    /// Creates a new unique [`StoreId`]
52    #[allow(clippy::new_without_default)] // reason = "StoreId::default() might be misunderstood to be some
53                                          // default value. However, a default value does not exist in that
54                                          // sense because every newly created StoreId must be unique. Also
55                                          // we don't want to allow the user to create new instances of
56                                          // this object."
57    pub(crate) fn new() -> Self {
58        static NEXT_STORE_ID: AtomicU64 = AtomicU64::new(0);
59
60        // TODO find a fix for the default wrapping behavior of `fetch_add`.
61        // Maybe we could return a `RuntimeError` here?
62        Self(NEXT_STORE_ID.fetch_add(1, Ordering::SeqCst))
63    }
64}

66/// A trait for types that might have a [`StoreId`] attached to them, so-called
67/// _stored_ types.
68pub trait AbstractStored: Sized {
69    type BareTy: Sized;
70
71    /// Creates a new stored object
72    ///
73    /// # Safety
74    ///
75    /// The caller has to guarantee that the bare value comes from a [`Store`]
76    /// with the given [`StoreId`].
77    unsafe fn from_bare(bare_value: Self::BareTy, id: StoreId) -> Self;
78
79    /// Converts this stored object into its bare form that does not have any [`StoreId`] attached to it.
80    fn into_bare(self) -> Self::BareTy;
81
82    /// Checks if this stored object comes from a specific store by its
83    /// [`StoreId`]. If true, it converts self into its bare form.
84    ///
85    /// # Panics
86    ///
87    /// This function panics in the case of mismatching store ids.
88    fn try_unwrap_into_bare(self, expected_store_id: StoreId) -> Self::BareTy;
89}
90
91impl<T: AbstractStored> AbstractStored for Vec<T> {
92    type BareTy = Vec<T::BareTy>;
93
94    /// Creates a new vector of stored objects from a vector of non-stored
95    /// objects.
96    ///
97    /// # Safety
98    ///
99    /// The caller has to guarantee that all bare values in the given vector come from a single
100    /// [`dlr_wasm_interpreter::Store`] with the given [`StoreId`].
101    unsafe fn from_bare(bare_value: Self::BareTy, id: StoreId) -> Self {
102        bare_value
103            .into_iter()
104            .map(|bare| {
105                // SAFETY: Upheld by caller
106                unsafe { T::from_bare(bare, id) }
107            })
108            .collect()
109    }
110
111    fn into_bare(self) -> Self::BareTy {
112        self.into_iter().map(T::into_bare).collect()
113    }
114
115    fn try_unwrap_into_bare(self, expected_store_id: StoreId) -> Self::BareTy {
116        self.into_iter()
117            .map(|t| t.try_unwrap_into_bare(expected_store_id))
118            .collect()
119    }
120}
