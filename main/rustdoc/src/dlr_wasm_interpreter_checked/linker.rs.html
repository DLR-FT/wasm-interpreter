11#[derive(Default)]
12pub struct Linker {
13    inner: dlr_wasm_interpreter_linker::Linker,
14
15    /// This is for the checked API which makes sure that all objects used
16    /// originate from the same [`Store`].
17    ///
18    /// Initially the store id is `None`. Only when a store-specific object or a
19    /// [`Store`] itself is used with a checked method, is this field set.  Once
20    /// initialized it is never written to again.
21    store_id: Option<StoreId>,
22}
23
24// All functions in this impl block must occur in the same order as they are
25// defined in for the unchecked [`Linker`] methods. Also all functions must
26// follow the same implementation scheme to make sure they are only light
27// wrappers:
28//
29// 1. get or insert the `StoreId` [of the store associated with the current `Linker`]
30// 2. try unwrap [stored parameter objects]
31// 3. call [unchecked method]
32// 4. rewrap [results into stored objects]
33// 5. return [stored result objects]
34impl Linker {
35    /// Creates a new [`Linker`] that is not yet associated to any specific [`Store`].
36    pub fn new() -> Self {
37        Self::default()
38    }
39
40    /// This is a safe variant of [`Linker::define`](dlr_wasm_interpreter_linker::Linker::define).
41    pub fn define(
42        &mut self,
43        module_name: String,
44        name: String,
45        extern_val: StoredExternVal,
46    ) -> Result<(), RuntimeError> {
47        // 1. get or insert the `StoreId`
48        let extern_val_store_id = extern_val.id();
49        let linker_store_id = *self.store_id.get_or_insert(extern_val_store_id);
50        if linker_store_id != extern_val_store_id {
51            panic!("Store id mismatch");
52        }
53        // 2. try unwrap
54        let extern_val = extern_val.try_unwrap_into_bare(linker_store_id);
55        // 3. call
56        // SAFETY: It was just checked that the `ExternVal` came from the store
57        // with the same id that is cached in the current linker instance.
58        unsafe { self.inner.define(module_name, name, extern_val) }?;
59        // 4. rewrap
60        // result is the unit type.
61        // 5. return
62        Ok(())
63    }
64
65    /// This is a safe variant of
66    /// [`Linker::define_module_instance`](dlr_wasm_interpreter_linker::Linker::define_module_instance).
67    pub fn define_module_instance<T: Config>(
68        &mut self,
69        store: &Store<T>,
70        module_name: String,
71        module: Stored<ModuleAddr>,
72    ) -> Result<(), RuntimeError> {
73        // 1. get or insert the `StoreId`
74        let module_store_id = module.id();
75        let linker_store_id = *self.store_id.get_or_insert(module_store_id);
76        if linker_store_id != module_store_id {
77            panic!("Store id mismatch");
78        }
79        // 2. try unwrap
80        let module = module.try_unwrap_into_bare(linker_store_id);
81        // 3. call
82        // SAFETY: It was just checked that the `ExternVal` came from the store
83        // with the same id that is cached in the current linker instance.
84        unsafe {
85            self.inner
86                .define_module_instance(store.inner(), module_name, module)
87        }?;
88        // 4. rewrap
89        // result is the unit type.
90        // 5. return
91        Ok(())
92    }

94    /// This is a safe variant of [`Linker::get`](dlr_wasm_interpreter_linker::Linker::get).
95    pub fn get(&self, module_name: String, name: String) -> Option<StoredExternVal> {
96        // 1. get or insert the `StoreId`
97        // Note: We can only get the id. If it has not been set yet, no
98        // definitions could have been made.
99        let linker_store_id = self.store_id?;
100        // 2. try unwrap
101        // no stored parameters
102        // 3. call
103        let extern_val = self.inner.get(module_name, name)?;
104        // 4. rewrap
105        // SAFETY: The `ExternVal` just came from the current `Linker`. Because
106        // a `Linker` can always be used with only one unique `Store`, this
107        // `ExternVal` must be from the current Linker's store.
108        let stored_extern_val = unsafe { StoredExternVal::from_bare(extern_val, linker_store_id) };
109        // 5. return
110        Some(stored_extern_val)
111    }
112
113    /// This is a variant of
114    /// [`Linker::instantiate_pre`](dlr_wasm_interpreter_linker::Linker::instantiate_pre).
115    pub fn instantiate_pre(&self, module: &Module) -> Option<Vec<StoredExternVal>> {
116        // Special case: If the module has no imports, we don't perform any
117        // linking. We need this special case, so that a `Linker`, that has not
118        // yet been associated with some `Store`, can still be used to
119        // pre-instantiate modules.
120        if module.imports().len() == 0 {
121            return Some(Vec::new());
122        }
123        // 1. get or insert `StoreId`
124        // Note: We can only get the id. If it has not been set yet, no
125        // definitions could have been made.
126        let linker_store_id = self.store_id?;
127        // 2. try unwrap
128        // no stored parameters
129        // 3. call
130        let extern_vals = self.inner.instantiate_pre(module)?;
131        // 4. rewrap
132        // SAFETY: All `ExternVal`s just came from the current `Linker`. Because
133        // a Linker can always be used with only one unique `Store`, all
134        // `ExternVal`s must be from the current Linker's store.
135        let stored_extern_vals = unsafe { Vec::from_bare(extern_vals, linker_store_id) };
136        // 5. retur
137        Some(stored_extern_vals)
138    }

140    /// This is a safe variant of
141    /// [`Linker::module_instantiate`](dlr_wasm_interpreter_linker::Linker::module_instantiate).
142    pub fn module_instantiate<'b, T: Config>(
143        &mut self,
144        store: &mut Store<'b, T>,
145        module: &Module<'b>,
146        maybe_fuel: Option<u64>,
147    ) -> Option<Result<StoredInstantiationOutcome, RuntimeError>> {
148        // 1. get or insert `StoreId`
149        let linker_store_id = *self.store_id.get_or_insert(store.id);
150        if linker_store_id != store.id {
151            panic!("Store id mismatch");
152        }
153        // 2. try unwrap
154        // no stored parameters
155        // 3. call
156        // SAFETY: It was just checked that the `ExternVal` came from the store
157        // with the same id that is cached in the current linker instance.
158        let instantiation_outcome = match unsafe {
159            self.inner
160                .module_instantiate(&mut store.inner, module, maybe_fuel)
161        } {
162            Some(Ok(instantiation_outcome)) => instantiation_outcome,
163            Some(Err(err)) => return Some(Err(err)),
164            None => return None,
165        };
166        // 4. rewrap
167        // SAFETY: The `InstantiationOutcome` just came from the current
168        // `Linker`. Because a linker can always be used with only one unique
169        // `Store`, the `InstantiationOutcome` must be from the current Linker's
170        // store.
171        let stored_instantiation_outcome = unsafe {
172            StoredInstantiationOutcome::from_bare(instantiation_outcome, linker_store_id)
173        };
174        // 5. return
175        Some(Ok(stored_instantiation_outcome))
176    }
