5/// The operation would remove more elements than currently present or tries to access an element
6/// when none are present
7#[derive(Debug, PartialEq, Eq, Clone)]
8pub struct EmptyContainerError;
9
10/// The operation would add elements past the maximum capacity
11#[derive(Debug, PartialEq, Eq, Clone)]
12pub struct FullContainerError;
13
14impl From<FullContainerError> for crate::RuntimeError {
15    fn from(_: FullContainerError) -> Self {
16        Self::StackExhaustion
17    }
18}

22pub struct FixedCapacityVec<T> {
23    /// A contiguous, non-reallocating, heap allocated vector. Behaves like a subset of
24    /// [`alloc::vec::Vec`], cleansed of any operation that reallocates. Backed by via boxed
25    /// slice containing elements of type `MaybeUninit<T>`. The maximum size (pendant to
26    /// [`alloc::vec::Vec::capacity`]) is determined at creation and remains unchanged over the
27    /// lifetime of [`Self`]. Nonetheless a variable amount of `T` can be kept within [`Self`], just
28    /// never more than the maximum size passed to the constructor.
29    ///
30    /// This can be used as a stack, in which case the first element (forming the bottom of the
31    /// stack) is at index 0, and new elements grow towards [`Self`]'s end. The last element
32    /// (`self.stack[self.stack.len() - 1]` is the topmost possible entry.
33    elements: Box<[MaybeUninit<T>]>,
34
35    /// Number of elements in [`Self`]. Also an index, pointing always to the first unused slot in
36    /// [`Self::elements`].
37    // Developer notes:
38    // - It is UB to access `self.elements[self.len]` or any other index larger than `self.len`.
39    // - If `self.stack_height == 0`, then there are no elements.
40    // - The guaranteed to be initialized range of [`Self::stack`] always is `self.stack[..self.stack_height]`.
41    len: usize,
42}
43
44impl<T> FixedCapacityVec<T> {
45    /// Construct new [`Self`], holding up to `capacity` elements of type `T`
46    pub fn with_capacity(capacity: usize) -> Self {
47        Self {
48            elements: Box::new_uninit_slice(capacity),
49            len: 0,
50        }
51    }
52
53    /// Check if the [`Self`] is empty
54    #[inline(always)]
55    pub const fn is_empty(&self) -> bool {
56        self.len == 0
57    }
58
59    /// Check if the [`Self`] is full
60    #[inline(always)]
61    pub const fn is_full(&self) -> bool {
62        self.len == self.elements.len()
63    }
64
65    /// Get the [`Self`] is height.
66    #[inline(always)]
67    pub const fn len(&self) -> usize {
68        self.len
69    }
70
71    /// Get the maximum number of elements that fit into [`Self`].
72    #[inline(always)]
73    pub const fn capacity(&self) -> usize {
74        self.elements.len()
75    }
76
77    /// Push a value to the end of [`Self`]
78    ///
79    /// # Safety
80    ///
81    /// - Causes UB if [`Self`] is already full.
82    /// - Causes UB if [`Self`] has a capacity equal to `usize::MAX`
83    #[inline(always)]
84    pub unsafe fn push_unchecked(&mut self, value: T) {
85        debug_assert!(!self.is_full());
86        debug_assert!(self.capacity() < usize::MAX);
87
88        // SAFETY: This must never be called when `Self` is full
89        unsafe { *self.elements.get_unchecked_mut(self.len) = MaybeUninit::new(value) };
90
91        // SAFETY: This must never be called when `self.capacity()` is `usize::MAX`
92        self.len = unsafe { self.len.unchecked_add(1) };
93    }
94
95    /// Push a value to the end of [`Self`]
96    pub fn push(&mut self, value: T) -> Result<(), FullContainerError> {
97        // check if insertion will overflow `self.len`
98        let new_len = self.len.checked_add(1).ok_or(FullContainerError)?;
99
100        *self.elements.get_mut(self.len).ok_or(FullContainerError)? = MaybeUninit::new(value);
101
102        // only "commit" the `new_len` if the insertion actually succeeded
103        self.len = new_len;
104
105        Ok(())
106    }
107
108    /// Pop a value from [`Self`]'s top/tail
109    ///
110    /// # Safety
111    ///
112    /// Causes UB if the stack is empty.
113    #[inline(always)]
114    pub unsafe fn pop_unchecked(&mut self) -> T {
115        debug_assert!(!self.is_empty());
116
117        // SAFETY: This must never be called when `self` is empty. If the stack is indeed not
118        // empty, then both the unchecked subtraction is guaranteed not to underflow.
119        self.len = unsafe { self.len.unchecked_sub(1) };
120
121        // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the number
122        // of initialized values starting from `self.element`'s beginning. As we require `self.len`
123        // not to be zero before calling this function, it is now at zero and thus a valid index to
124        // an initialized element.
125        unsafe { self.elements.get_unchecked(self.len).assume_init_read() }
126    }
127
128    /// Pop a value from [`Self`]'s top/tail
129    pub fn pop(&mut self) -> Result<T, EmptyContainerError> {
130        self.len = self.len.checked_sub(1).ok_or(EmptyContainerError)?;
131
132        // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the number
133        // of initialized values starting from `self.element`'s beginning. As we require `self.len`
134        // not to be zero before calling this function, it is now at zero and thus a valid index to
135        // an initialized element.
136        let value = unsafe { self.elements.get_unchecked(self.len).assume_init_read() };
137
138        Ok(value)
139    }
140
141    /// Peek at the topmost/last value from [`Self`]
142    ///
143    /// # Safety
144    ///
145    /// Causes UB if the stack is empty.
146    #[inline(always)]
147    pub unsafe fn peek_unchecked(&self) -> &T {
148        debug_assert!(!self.is_empty());
149
150        // SAFETY: This must never be called when the `self` is empty. If the stack is indeed not
151        // empty, then both the unchecked subtraction can not underflow, and the element at that
152        // index will be already initialized.
153        unsafe {
154            self.elements
155                .get_unchecked(self.len.unchecked_sub(1))
156                .assume_init_ref()
157        }
158    }
159
160    /// Peek at the topmost/last value from [`Self`]
161    pub fn peek(&self) -> Result<&T, EmptyContainerError> {
162        let idx = self.len.checked_sub(1).ok_or(EmptyContainerError)?;
163
164        // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the number
165        // of initialized values starting from `self.element`'s beginning. As we require `self.len`
166        // not to be zero before calling this function, it is now at zero and thus a valid index to
167        // an initialized element.
168        let value = unsafe { self.elements.get_unchecked(idx).assume_init_ref() };
169
170        Ok(value)
171    }
172
173    /// Get a shared ref to the nth element in [`Self`]
174    #[inline(always)]
175    pub fn get(&self, idx: usize) -> Option<&T> {
176        if idx < self.len {
177            self.elements.get(idx).map(|e| {
178                // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the
179                // number of initialized values starting from `self.element`'s beginning. The pervious
180                // if condition in term checks that idx is Smaller than
181                unsafe { e.assume_init_ref() }
182            })
183        } else {
184            None
185        }
186    }
187
188    /// Get a mut ref to the nth element in [`Self`]
189    #[inline(always)]
190    pub fn get_mut(&mut self, idx: usize) -> Option<&mut T> {
191        if idx < self.len {
192            self.elements.get_mut(idx).map(|e| {
193                // SAFETY: per the previous if statement, the index points into the range of
194                // initialized stack members.
195                unsafe { e.assume_init_mut() }
196            })
197        } else {
198            None
199        }
200    }
201}

203impl<T: Clone> FixedCapacityVec<T> {
204    /// Push from a slice into [`Self`], appending after the last/topmost element
205    pub fn push_from_slice(&mut self, values: &[T]) -> Result<(), FullContainerError> {
206        // verify `values` fits into the new self
207        if values.len() > (self.elements.len() - self.len) {
208            return Err(FullContainerError);
209        }
210
211        for e in values {
212            // SAFETY: this is safe, as the previous if statement verifies that there is enough
213            // space for all elements from `values` in `stack`
214            unsafe { self.push_unchecked(e.clone()) };
215        }
216
217        Ok(())
218    }
219
220    /// Pop `n` elements from [`Self`] into a slice. The topmost/last element of [`Self`] will
221    /// become the slice's last element.
222    pub fn pop_into_slice(
223        &mut self,
224        n: usize,
225    ) -> Result<impl core::ops::Deref<Target = [T]> + '_, EmptyContainerError> {
226        // verify at least `n` elements are still one the stack
227        if n > self.elements.len() {
228            return Err(EmptyContainerError);
229        }
230
231        let old_len = self.len();
232
233        // SAFETY: the previous if statement ensure that `n` is smaller than or equal to
234        // `self.elements.len()`, hence this subtraction is guaranteed to not underflow
235        self.len = unsafe { self.len.unchecked_sub(n) };
236
237        // SAFETY: the prior if statement checks that `stack_height - n` wont underflow. And the
238        // rest of the module ensures that stack_height never gets bigger than `self.stack.len()`.
239        // Thus the index range is always in-bounds.
240        Ok(SliceDropGuard(unsafe {
241            self.elements.get_unchecked_mut(self.len..old_len)
242        }))
243    }
244}

246pub struct SliceDropGuard<'a, T>(&'a mut [MaybeUninit<T>]);
247
248impl<'a, T> core::ops::Deref for SliceDropGuard<'a, T> {
249    type Target = [T];
250
251    fn deref(&self) -> &Self::Target {
252        // SAFETY: The rest of the module ensures that stack_height never gets bigger than
253        // `self.stack.len()`. Thus the index range is always in-bounds.
254        unsafe { slice_assume_init(self.0) }
255    }
256}
257
258impl<'a, T> Drop for SliceDropGuard<'a, T> {
259    fn drop(&mut self) {
260        for e in &mut *self.0 {
261            // SAFETY: The rest of the module ensures that stack_height never gets bigger than
262            // `self.stack.len()`. Thus the index range is always in-bounds.
263            unsafe { e.assume_init_drop() };
264        }
265    }
266}

268impl<T: Copy> FixedCapacityVec<T> {
269    /// Remove `remove_count` values from [`Self`], keeping the topmost `keep_count` values
270    ///
271    /// From the [`Self`], remove `remove_count` elements, by sliding down the `keep_count` last/topmost
272    /// values `remove_count` positions forward/towards the bottom.
273    ///
274    /// **Effects**
275    ///
276    /// - after the operation, [`Self`] will contain `remove_count` fewer elements
277    /// - `keep_count` topmost elements will be identical before and after the operation
278    /// - all elements below the `remove_count + keep_count` topmost stack entry remain
279    pub fn remove_in_between(&mut self, remove_count: usize, keep_count: usize) {
280        // TODO make unchecked version, remove overflowing arithmetic in safe version
281        let len = self.len();
282        self.elements
283            .copy_within(len - keep_count..len, len - keep_count - remove_count);
284        self.len -= remove_count;
285    }
286}

288impl<T: core::fmt::Debug> core::fmt::Debug for FixedCapacityVec<T> {
289    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
290        // SAFETY: per the guarantees of push/pop/etc., `0..self.stack_height` is always a range to
291        // initalized values.
292        let stack_elements_slice = unsafe { slice_assume_init(&self.elements[..self.len]) };
293
294        f.debug_struct("Stack")
295            .field("stack", &stack_elements_slice)
296            .field("stack_height", &self.len)
297            .finish()
298    }
299}

301impl<T> Drop for FixedCapacityVec<T> {
302    fn drop(&mut self) {
303        // SAFETY: self is guaranteed to have at least `self.len` elements, because `sel.len` can
304        // not grow past `self.elements.len` (aka `self.capacity()`).
305        for e in unsafe { self.elements.get_unchecked_mut(0..self.len) } {
306            // SAFETY: This is guaranteed to be initialized, as `self.len` accurately tracks the
307            // number of initialized values starting from `self.element`'s beginning.
308            unsafe { e.assume_init_drop() }
309        }
310    }

313// TODO use assume_init_ref, once we get the MSRV to 1.93.0
314/// # Safety
315///
316/// Use this to assume a range of a slice to be initialized. This is provided starting with Rust
317/// 1.93.0.
318#[inline(always)]
319unsafe fn slice_assume_init<T>(slice: &[MaybeUninit<T>]) -> &[T] {
320    // SAFETY: casting `slice` to a `*const [T]` is safe since the caller guarantees that
321    // `slice` is initialized, and `MaybeUninit` is guaranteed to have the same layout as `T`.
322    // The pointer obtained is valid since it refers to memory owned by `slice` which is a
323    // reference and thus guaranteed to be valid for reads.
324    unsafe { &*(slice as *const _ as *const [T]) }
