1use anyhow::Context as _;

8#[derive(Parser, Debug)]
9#[command(version, about, long_about = None)]
10/// Generate a Markdown report for the differences of testsuite results
11struct Args {
12    /// Path to the testsuite output JSON for the old version
13    old_results_path: PathBuf,
14
15    /// Path to the testsuite output JSON for the new version
16    new_results_path: PathBuf,
17}
18
19mod ci_reports;
20mod deltas;
21mod summary;
22
23fn main() -> anyhow::Result<()> {
24    // Read arguments
25    let args = Args::parse();
26    let old_file = File::open(args.old_results_path).context("failed to open old results file")?;
27    let new_file = File::open(args.new_results_path).context("failed to open new results file")?;
28
29    // Read & parse the data
30    let old_report: CIFullReport = serde_json::from_reader(BufReader::new(old_file))
31        .context("failed to parse old file contents")?;
32    let new_report: CIFullReport = serde_json::from_reader(BufReader::new(new_file))
33        .context("failed to parse new file contents")?;
34
35    // Analyze the data
36    let deltas =
37        deltas::generate(&old_report, &new_report).context("failed to generate file deltas")?;
38    let summary = summary::generate(&new_report);
39    let report = TestsuiteReport { deltas, summary };
40
41    // Generate the final report
42    println!("{report}");
43
44    Ok(())
45}

47struct TestsuiteReport<'report> {
48    deltas: FileDeltas<'report>,
49    summary: ReportSummary<'report>,
50}

52impl Display for TestsuiteReport<'_> {
53    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
54        writeln!(
55            f,
56            "# 🗒️ [WebAssembly Testsuite](https://github.com/WebAssembly/testsuite) Report"
57        )?;
58        writeln!(f)?;
59
60        writeln!(f, "{}", self.deltas)?;
61        writeln!(f, "{}", self.summary)
62    }
63}

69pub fn write_details_summary<W: std::fmt::Write>(
70    writer: &mut W,
71    summary: impl FnOnce(&mut W) -> std::fmt::Result,
72    contents: impl FnOnce(&mut W) -> std::fmt::Result,
