1//! A simple registry for managing host hunctions/calls.
2//!
3//! Contrary to the other crates, this crate does not provide a re-useable
4//! general-purpose implementation. Instead, this implementation is used for
5//! tests and the specificaiton testsuite runner.
6
7#![no_std]
8
9extern crate alloc;
10
11use alloc::{borrow::ToOwned, boxed::Box, vec::Vec};
12
13use dlr_wasm_interpreter::{
14    Config, FuncAddr, FuncType, HostResumable, ResultType, RuntimeError, ValueTypeMismatchError,
15};
16use dlr_wasm_interpreter_checked::{
17    Store, Stored, StoredHostCall, StoredInteropValueList, StoredRunState, StoredValue,
18};
19
20type BoxedHostFn<T> = Box<dyn FnMut(&mut T, Vec<StoredValue>) -> Vec<StoredValue>>;
21
22/// A simple registry for host functions.
23///
24/// # Features
25///
26/// - based on the [`checked API`](dlr_wasm_interpreter_checked) and its `interop` feature
27/// - host functions may access generic user data `T`
28/// - boxed and dynamically dispatched host functions
29pub struct Registry<T> {
30    host_functions: Vec<BoxedHostFn<T>>,
31    next_hostcode: usize,
32}
33
34impl<T> Default for Registry<T> {
35    fn default() -> Self {
36        Self {
37            host_functions: Vec::new(),
38            next_hostcode: 0,
39        }
40    }
41}
42
43impl<T> Registry<T> {
44    pub fn alloc_host_function<C: Config>(
45        &mut self,
46        store: &mut Store<C>,
47        func_type: FuncType,
48        host_function: impl FnMut(&mut T, Vec<StoredValue>) -> Vec<StoredValue> + 'static,
49    ) -> Stored<FuncAddr> {
50        let hostcode = self.next_hostcode;
51        self.next_hostcode += 1;
52
53        self.host_functions.push(Box::new(host_function));
54
55        store.func_alloc(func_type, hostcode)
56    }
57
58    pub fn alloc_host_function_typed<C, Params, Returns, F>(
59        &mut self,
60        store: &mut Store<C>,
61        mut host_function: F,
62    ) -> Stored<FuncAddr>
63    where
64        C: Config,
65        Params: StoredInteropValueList,
66        Returns: StoredInteropValueList,
67        F: FnMut(&mut T, Params) -> Returns + 'static,
68    {
69        let func_type = FuncType {
70            params: ResultType {
71                valtypes: Params::TYS.to_owned(),
72            },
73            returns: ResultType {
74                valtypes: Returns::TYS.to_owned(),
75            },
76        };
77
78        self.alloc_host_function(store, func_type, move |user_data, params| {
79            let params = Params::try_from_values(params.into_iter()).unwrap();
80            let returns = host_function(user_data, params);
81            returns.into_values()
82        })
83    }
84
85    pub fn invoke_without_fuel<C: Config>(
86        &mut self,
87        user_data: &mut T,
88        store: &mut Store<C>,
89        func_addr: Stored<FuncAddr>,
90        params: Vec<StoredValue>,
91    ) -> Result<Vec<StoredValue>, RuntimeError> {
92        let resumable = store.create_resumable(func_addr, params, None)?;
93        let mut run_state = store.resume(resumable)?;
94        loop {
95            match run_state {
96                StoredRunState::Finished { values, .. } => return Ok(values),
97                StoredRunState::Resumable {
98                    resumable,
99                    required_fuel,
100                } => {
101                    assert!(required_fuel.is_none(), "fuel is disabled");
102                    run_state = store.resume_wasm(resumable)?;
103                }
104                StoredRunState::HostCalled {
105                    host_call,
106                    resumable,
107                } => {
108                    run_state = self.perform_host_call(user_data, store, host_call, resumable)?;
109                }
110            }
111        }
112    }
113
114    pub fn invoke_without_fuel_typed<C, Params, Returns>(
115        &mut self,
116        user_data: &mut T,
117        store: &mut Store<C>,
118        func_addr: Stored<FuncAddr>,
119        params: Params,
120    ) -> Result<Returns, RuntimeError>
121    where
122        C: Config,
123        Params: StoredInteropValueList,
124        Returns: StoredInteropValueList,
125    {
126        let params = params.into_values();
127        let returns = self.invoke_without_fuel(user_data, store, func_addr, params)?;
128        Returns::try_from_values(returns.into_iter())
129            .map_err(|ValueTypeMismatchError| RuntimeError::FunctionInvocationSignatureMismatch)
130    }
131
132    pub fn perform_host_call<C: Config>(
133        &mut self,
134        user_data: &mut T,
135        store: &mut Store<C>,
136        host_call: StoredHostCall,
137        host_resumable: Stored<HostResumable>,
138    ) -> Result<StoredRunState, RuntimeError> {
139        let host_function = &mut self.host_functions[host_call.hostcode];
140        let returns = host_function(user_data, host_call.params);
141        store.finish_host_call(host_resumable, returns)
142    }
143}
